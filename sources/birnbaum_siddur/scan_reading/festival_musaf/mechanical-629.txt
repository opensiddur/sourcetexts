אִם טוֹבִים הֵם, חַזְּקֵם וְאַמְּצֵם, וְיִתְקַיְּמוּ בִי וּבָהֶם כַּחֲלוֹמוֹת שֶׁל יוֹסֵף הַצַּדִּיק; וְאִם צְרִיכִים רְפוּאָה, רְפָאֵם כְּחִזְקִיָּֽהוּ מֶֽלֶךְ יְהוּדָה מֵחׇלְיוֹ, וּכְמִרְיָם הַנְּבִיאָה מִצָּרַעְתָּהּ, וּכְנַעֲמָן מִצָּרַעְתּוֹ, וּכְמֵי מָרָה עַל יְדֵי משֶׁה רַבֵּֽנוּ, וּכְמֵי יְרִיחוֹ עַל יְדֵי אֱלִישָׁע. וּכְשֵׁם שֶׁהָפַֽכְתָּ אֶת קִלְלַת בִּלְעָם הָרָשָׁע מִקְּלָלָה לִבְרָכָה, כֵּן תַּהֲפֹךְ כׇּל חֲלוֹמוֹתַי עָלַי וְעַל כׇּל יִשְׂרָאֵל לְטוֹבָה, וְתִשְׁמְרֵֽנִי וּתְחׇנֵּֽנִי וְתִרְצֵֽנִי. אָמֵן.
-----
יָאֵר
אֱלֹהִים יְחׇנֵּֽנוּ וִיבָרְכֵֽנוּ; יָאֵר פָּנָיו אִתָּֽנוּ סֶֽלָה.
יְהֹוָה
יְיָ יְיָ, אֵל רַחוּם וְחַנּוּן, אֶֽרֶךְ אַפַּֽיִם וְרַב חֶֽסֶד וֶאֱמֶת.
פָּנָיו
פְּנֵה אֵלַי וְחׇנֵּנִי, כִּי יָחִיד וְעָנִי אָֽנִי.
אֵלֶֽיךָ
אֵלֶֽיךָ, יְיָ, נַפְשִׁי אֶשָּׂא.
וִיחֻנֶּֽךָ.
הִנֵּה כְעֵינֵי עֲבָדִים אֶל יַד אֲדוֹנֵיהֶם, כְּעֵינֵי שִׁפְחָה אֶל יַד גְּבִרְתָּהּ, כֵּן עֵינֵֽינוּ אֶל יְיָ אֱלֹהֵֽינוּ עַד שֶׁיְּחׇנֵּֽנוּ.
-----
יִשָּׂא
יִשָּׂא בְרָכָה מֵאֵת יְיָ, וּצְדָקָה מֵאֱלֹהֵי יִשְׁעוֹ. וּמְצָא חֵן וְשֵֽׂכֶל טוֹב בְּעֵינֵי אֱלֹהִים וְאָדָם.
יְהֹוָה
יְיָ, חׇנֵּֽנוּ, לְךָ קִוִּֽינוּ, הֱיֵה זְרֹעָם לַבְּקָרִים, אַף יְשׁוּעָתֵֽנוּ בְּעֵת צָרָה.
פָּנָיו
אַל תַּסְתֵּר פָּנֶֽיךָ מִמֶּֽנִּי בְּיוֹם צַר לִי; הַטֵּה אֵלַי אׇזְנֶֽךָ, בְּיוֹם אֶקְרָא מַהֵר עֲנֵֽנִי.
אֵלֶֽיךָ
אֵלֶֽיךָ נָשָֽׂאתִי אֶת עֵינַי, הַיֹּשְׁבִי בַּשָּׁמָֽיִם.
וְיָשֵׂם
וְשָׂמוּ אֶת שְׁמִי עַל בְּנֵי יִשְׂרָאֵל, וַאֲנִי אֲבָרְכֵֽם.
לְךָ
לְךָ, יְיָ, הַגְּדֻלָּה וְהַגְּבוּרָה וְהַתִּפְאֶֽרֶת וְהַנֵּֽצַח וְהַהוֹד, כִּי כֹל בַּשָּׁמַֽיִם וּבָאָֽרֶץ; לְךָ, יְיָ, הַמַּמְלָכָה וְהַמִּתְנַשֵּׂא לְכֹל לְרֹאשׁ.
שָׁלוֹם.
שָׁלוֹם שָׁלוֹם לָרָחוֹק וְלַקָּרוֹב, אָמַר יְיָ, וּרְפָאתִיו.
