יְיָ לְעוֹלָם תַּעֲמֹד, מַחְשְׁבוֹת לִבּוֹ לְדֹר וָדֹר. אַשְׁרֵי הַגּוֹי אֲשֶׁר יְיָ אֱלֹהָיו, הָעָם בָּחַר לְנַחֲלָה לוֹ. מִשָּׁמַֽיִם הִבִּיט יְיָ, רָאָה אֶת כָּל בְּנֵי הָאָדָם. מִמְּכוֹן שִׁבְתּוֹ הִשְׁגִּֽיחַ, אֶל כָּל יֹשְׁבֵי הָאָֽרֶץ. הַיֹּצֵר יַֽחַד לִבָּם, הַמֵּבִין אֶל כָּל מַעֲשֵׂיהֶם. אֵין הַמֶּֽלֶךְ נוֹשָׁע בְּרָב חָֽיִל, גִּבּוֹר לֹא יִנָּצֵל בְּרָב כֹּֽחַ. שֶֽׁקֶר הַסּוּס לִתְשׁוּעָה, וּבְרֹב חֵילוֹ לֹא יְמַלֵּט. הִנֵּה עֵין יְיָ אֶל יְרֵאָיו, לַמְיַחֲלִים לְחַסְדּוֹ. לְהַצִּיל מִמָּֽוֶת נַפְשָׁם, וּלְחַיּוֹתָם בָּרָעָב. נַפְשֵׁנוּ חִכְּתָה לַיְיָ, עֶזְרֵֽנוּ וּמָגִנֵּֽנוּ הוּא. כִּי בוֹ יִשְׂמַח לִבֵּֽנוּ, כִּי בְשֵׁם קָדְשׁוֹ בָטָֽחְנוּ. יְהִי חַסְדְּךָ יְיָ עָלֵֽינוּ, כַּאֲשֶׁר יִחַֽלְנוּ לָךְ.

לֹא יָרֵֽעוּ וְלֹא יַשְׁחִֽיתוּ בְּכָל הַר קָדְשִׁי, כִּי מָלְאָה הָאָֽרֶץ דֵּעָה אֶת יְיָ, כַּמַּֽיִם לַיָּם מְכַסִּים.

שִׁיר הַמַּעֲלוֹת. מִמַּעֲמַקִּים קְרָאתִֽיךָ, יְיָ. אֲדֹנָי, שִׁמְעָה בְקוֹלִי, תִּהְיֶֽינָה אָזְנֶֽיךָ קַשֻּׁבוֹת לְקוֹל תַּחֲנוּנָי. אִם עֲוֺנוֹת תִּשְׁמָר־יָהּ, אֲדֹנָי מִי יַעֲמֹד. כִּי עִמְּךָ הַסְּלִיחָה, לְמַֽעַן תִּוָּרֵא. קִוִּֽיתִי יְיָ, קִוְּתָה נַפְשִׁי, וְלִדְבָרוֹ הוֹחָֽלְתִּי. נַפְשִׁי לַאדֹנָי מִשֹּׁמְרִים לַבֹּֽקֶר, שֹׁמְרִים לַבֹּֽקֶר. יַחֵל יִשְׂרָאֵל אֶל יְיָ, כִּי עִם יְיָ הַחֶֽסֶד, וְהַרְבֵּה עִמּוֹ פְדוּת. וְהוּא יִפְדֶּה אֶת יִשְׂרָאֵל מִכֹּל עֲוֺנֹתָיו.
