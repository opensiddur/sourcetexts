יְהִי רָצוֹן מִלְּפָנֶֽיךָ, יְיָ אֱלֹהַי וֵאלֹהֵי אֲבוֹתַי, שֶׁתַּשְׁרֶה שְׁכִינָתְךָ בֵּינֵֽינוּ, וְתִפְרוֹשׂ עָלֵֽינוּ סֻכַּת שְׁלוֹמֶֽךָ, וְתַקִּיף אוֹתָֽנוּ מִזִּיו כְּבוֹדָךְ הַקָּדוֹשׁ וְהַטָּהוֹר. וְלָרְעֵבִים גַּם צְמֵאִים, תֵּן לַחְמָם וּמַֽיִם נֶאֱמָנִים. וּתְזַכֵּֽנוּ לֵשֵׁב יָמִים רַבִּים עַל הָאֲדָמָה, אַדְמַת קֹֽדֶשׁ, בַּעֲבוֹדָתְךָ וּבְיִרְאָתֶֽךָ. בָּרוּךְ יְיָ לְעוֹלָם, אָמֵן וְאָמֵן.

אֲזַמֵּן לִסְעוּדָתִי אֻשְׁפִּיזִין עִלָּאִין: אַבְרָהָם יִצְחָק וְיַעֲקֹב, יוֹסֵף משֶׁה אַהֲרֹן וְדָוִד.

בְּמָטוּ מִנָּךְ, אַבְרָהָם אֻשְׁפִּיזִי עִלָּאִי, דְּיֵתְבוּ עִמִּי וְעִמָּךְ כׇּל אֻשְׁפִּיזֵי עִלָּאֵי: יִצְחָק יַעֲקֹב יוֹסֵף, משֶׁה אַהֲרֹן וְדָוִד.

בְּמָטוּ מִנָּךְ, יִצְחָק אֻשְׁפִּיזִי עִלָּאִי, דְּיֵתְבוּ עִמִּי וְעִמָּךְ כׇּל אֻשְׁפִּיזֵי עִלָּאֵי: אַבְרָהָם יַעֲקֹב יוֹסֵף, משֶׁה אַהֲרֹן יוֹסֵף וְדָוִד.

בְּמָטוּ מִנָּךְ, יַעֲקֹב אֻשְׁפִּיזִי עִלָּאִי, דְּיֵתְבוּ עִמִּי וְעִמָּךְ כׇּל אֻשְׁפִּיזֵי עִלָּאֵי: אַבְרָהָם יִצְחָק יוֹסֵף, משֶׁה אַהֲרֹן וְדָוִד.

בְּמָטוּ מִנָּךְ, יוֹסֵף אֻשְׁפִּיזִי עִלָּאִי, דְּיֵתְבוּ עִמִּי וְעִמָּךְ כׇּל אֻשְׁפִּיזֵי עִלָּאֵי: אַבְרָהָם יִצְחָק יַעֲקֹב, משֶׁה אַהֲרֹן וְדָוִד.

בְּמָטוּ מִנָּךְ, משֶׁה אֻשְׁפִּיזִי עִלָּאִי, דְּיֵתְבוּ עִמִּי וְעִמָּךְ כׇּל אֻשְׁפִּיזֵי עִלָּאֵי: אַבְרָהָם יִצְחָק יַעֲקֹב, יוֹסֵף אַהֲרֹן וְדָוִד.
