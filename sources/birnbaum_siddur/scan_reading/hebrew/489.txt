אָמַר לָהֶם: צְאוּ וּרְאוּ אֵיזוֹ הִיא דֶֽרֶךְ רָעָה שֶׁיִּתְרַחֵק מִמֶּֽנָּה הָאָדָם. רַבִּי אֱלִיעֶֽזֶר אוֹמֵר: עַֽיִן רָעָה. רַבִּי יְהוֹשֻֽׁעַ אוֹמֵר: חָבֵר רָע. רַבִּי יוֹסֵי אוֹמֵר שָׁכֵן רָע. רַבִּי שִׁמְעוֹן אוֹמֵר: הַלֹּוֶה וְאֵינוֹ מְשַׁלֵּם; אֶחָד הַלֹּוֶה מִן הָאָדָם כְּלֹוֶה מִן הַמָּקוֹם, שֶׁנֶּאֱמַר: לֹוֶה רָשָׁע וְלֹא יְשַׁלֵּם, וְצַדִּיק חוֹנֵן וְנוֹתֵן. רַבִּי אֶלְעָזָר אוֹמֵר: לֵב רָע. אָמַר לָהֶם: רוֹאֶה אֲנִי אֶת דִּבְרֵי אֶלְעָזָר בֶּן עֲרָךְ מִדִּבְרֵיכֶם, שֶׁבִּכְלַל דְּבָרָיו דִּבְרֵיכֶם.
הֵם אָמְרוּ שְׁלֹשָׁה דְבָרִים. רַבִּי אֱלִיעֶֽזֶר אוֹמֵר: יְהִי כְבוֹד חֲבֵרְךָ חָבִיב עָלֶֽיךָ כְּשֶׁלָּךְ, וְאַל תְּהִי נֽוֹחַ לִכְעוֹס, וְשׁוּב יוֹם אֶחָד לִפְנֵי מִיתָתְךָ. וֶהֱוֵי מִתְחַמֵּם כְּנֶֽגֶד אוּרָן שֶׁל חֲכָמִים, וֶהֱוֵי זָהִיר בְּגַחַלְתָּן שֶׁלֹּא תִכָּוֶה, שֶׁנְּשִׁיכָתָן נְשִׁיכַת שׁוּעָל, וַעֲקִיצָתָן עֲקִיצַת עַקְרָב, וּלְחִישָׁתָן לְחִישַׁת שָׂרָף, וְכָל דִּבְרֵיהֶם כְּגַחֲלֵי אֵשׁ.
רַבִּי יְהוֹשֻֽׁעַ אוֹמֵר: עֵין הָרָע וְיֵֽצֶר הָרָע וְשִׂנְאַת הַבְּרִיּוֹת מוֹצִיאִים אֶת הָאָדָם מִן הָעוֹלָם.
רַבִּי יוֹסֵי אוֹמֵר: יְהִי מָמוֹן חֲבֵרְךָ חָבִיב עָלֶֽיךָ כְּשֶׁלָּךְ, וְהַתְקֵן עַצְמְךָ לִלְמֹד תּוֹרָה שֶׁאֵינָהּ יְרֻשָּׁה לָךְ, וְכָל מַעֲשֶֽׂיךָ יִהְיוּ לְשֵׁם שָׁמַֽיִם.
רַבִּי שִׁמְעוֹן אוֹמֵר: הֱוֵי זָהִיר בִּקְרִיאַת שְׁמַע וּבִתְפִלָּה; וּכְשֶׁאַתָּה מִתְפַּלֵּל, אַל תַּֽעַשׂ תְּפִלָּתְךָ קֶֽבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים לִפְנֵי הַמָּקוֹם, שֶׁנֶּאֱמַר: כִּי חַנּוּן וְרַחוּם הוּא, אֶֽרֶךְ אַפַּיִם וְרַב חֶֽסֶד וְנִחָם עַל הָרָעָה; וְאַל תְּהִי רָשָׁע בִּפְנֵי עַצְמֶֽךָ.
