דָּוִד מֶֽלֶךְ יִשְׂרָאֵל חַי וְקַיָּם.

שָׁלוֹם עֲלֵיכֶם. עֲלֵיכֶם שָׁלוֹם.

סִמָּן טוֹב וּמַזָּל טוֹב יְהֵא לָנוּ וּלְכָל יִשְׂרָאֵל, אָמֵן.

קוֹל דּוֹדִי הִנֵּה זֶה בָּא, מְדַלֵּג עַל הֶהָרִים, מְקַפֵּץ עַל הַגְּבָעוֹת. דּוֹמֶה דוֹדִי לִצְבִי אוֹ לְעֹפֶר הָאַיָּלִים; הִנֵּה זֶה עוֹמֵד אַחַר כָּתְלֵנוּ, מַשְׁגִּיחַ מִן הַחַלֹּנוֹת, מֵצִיץ מִן הַחֲרַכִּים.

שִׁיר לַמַּעֲלוֹת. אֶשָּׂא עֵינַי אֶל הֶהָרִים, מֵאַיִן יָבֹא עֶזְרִי. עֶזְרִי מֵעִם יְיָ, עֹשֵׂה שָׁמַיִם וָאָרֶץ. אַל יִתֵּן לַמּוֹט רַגְלֶךָ, אַל יָנוּם שֹׁמְרֶךָ. הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל. יְיָ שֹׁמְרֶךָ, יְיָ צִלְּךָ, עַל יַד יְמִינֶךָ. יוֹמָם הַשֶּׁמֶשׁ לֹא יַכֶּכָּה, וְיָרֵחַ בַּלָּֽיְלָה. יְיָ יִשְׁמָרְךָ מִכָּל רָע, יִשְׁמֹר אֶת נַפְשֶׁךָ. יְיָ יִשְׁמָר צֵאתְךָ וּבוֹאֶךָ, מֵעַתָּה וְעַד עוֹלָם.

הַלְלוּיָהּ; הַלְלוּ אֵל בְּקָדְשׁוֹ, הַלְלֽוּהוּ בִּרְקִיעַ עֻזּוֹ. הַלְלֽוּהוּ בִגְבוּרֹתָיו; הַלְלֽוּהוּ כְּרֹב גֻּדְלוֹ. הַלְלֽוּהוּ בְּתֵקַע שׁוֹפָר; הַלְלֽוּהוּ בְּנֵבֶל וְכִנּוֹר. הַלְלֽוּהוּ בְּתֹף וּמָחוֹל, הַלְלֽוּהוּ בְּמִנִּים וְעֻגָב. הַלְלֽוּהוּ בְּצִלְצְלֵי שָׁמַע, הַלְלֽוּהוּ בְּצִלְצְלֵי תְרוּעָה. כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ; הַלְלוּיָהּ.

תָּנָא דְבֵי רַבִּי יִשְׁמָעֵאל: אִלְמָלֵי לֹא זָכוּ יִשְׂרָאֵל אֶלָּא לְהַקְבִּיל פְּנֵי אֲבִיהֶם שֶׁבַּשָּׁמַיִם פַּעַם אַחַת בַּחֹדֶשׁ, דַּיָּם. אָמַר אַבַּיֵּי: הִלְכָּךְ, צָרִיךְ לְמֵימְרָא מְעֻמָּד.
