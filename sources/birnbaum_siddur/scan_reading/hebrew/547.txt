עָשָׂה הַיְשׁוּעָה הַגְּדוֹלָה הַזֹּאת בְּיִשְׂרָאֵל; חָלִֽילָה, חַי יְיָ אִם יִפֹּל מִשַּׂעֲרַת רֹאשׁוֹ אַֽרְצָה, כִּי עִם אֱלֹהִים עָשָׂה הַיּוֹם הַזֶּה; וַיִּפְדּוּ הָעָם אֶת יוֹנָתָן, וְלֹא מֵת. וּפְדוּיֵי יְיָ יְשֻׁבוּן, וּבָֽאוּ צִיּוֹן בְּרִנָּה, וְשִׂמְחַת עוֹלָם עַל רֹאשָׁם; שָׂשׂוֹן וְשִׂמְחָה יַשִּֽׂיגוּ, וְנָֽסוּ יָגוֹן וַאֲנָחָה. הָפַֽכְתָּ מִסְפְּדִי לְמָחוֹל לִי, פִּתַּֽחְתָּ שַׂקִּי וַתְּאַזְּרֵֽנִי שִׂמְחָה. וְלֹא אָבָה יְיָ אֱלֹהֶֽיךָ לִשְׁמֹֽעַ אֶל בִּלְעָם, וַיַּהֲפֹךְ יְיָ אֱלֹהֶֽיךָ לְּךָ אֶת הַקְּלָלָה לִבְרָכָה, כִּי אֲהֵבְךָ יְיָ אֱלֹהֶֽיךָ. אָז תִּשְׂמַח בְּתוּלָה בְּמָחוֹל, וּבַחֻרִים וּזְקֵנִים יַחְדָּו; וְהָפַכְתִּי אֶבְלָם לְשָׂשׂוֹן, וְנִחַמְתִּים וְשִׂמַּחְתִּים מִיגוֹנָם.

בּוֹרֵא נִיב שְׂפָתָֽיִם, שָׁלוֹם שָׁלוֹם לָרָחוֹק וְלַקָּרוֹב, אָמַר יְיָ וּרְפָאתִיו. וְרֽוּחַ לָבְשָׁה אֶת עֲמָשַׂי, רֹאשׁ הַשָּׁלִישִׁים; לְךָ דָוִיד, וְעִמְּךָ בֶן־יִשַׁי, שָׁלוֹם שָׁלוֹם לְךָ, וְשָׁלוֹם לְעֹזְרֶֽךָ, כִּי עֲזָרְךָ אֱלֹהֶֽיךָ; וַיְקַבְּלֵם דָּוִיד, וַיִּתְּנֵם בְּרָאשֵׁי הַגְּדוּד. וַאֲמַרְתֶּם כֹּה לֶחָי: וְאַתָּה שָׁלוֹם, וּבֵיתְךָ שָׁלוֹם, וְכֹל אֲשֶׁר לְךָ שָׁלוֹם. יְיָ עֹז לְעַמּוֹ יִתֵּן; יְיָ יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם.

אָמַר רַבִּי יוֹחָנָן: בְּכָל מָקוֹם שָׁאַתָּה מוֹצֵא גְדֻלָּתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, שָׁם אַתָּה מוֹצֵא עַנְוְתָנוּתוֹ. דָּבָר זֶה כָּתוּב בַּתּוֹרָה, וְשָׁנוּי בַּנְּבִיאִים, וּמְשֻׁלָּשׁ בַּכְּתוּבִים. כָּתוּב בַּתּוֹרָה: כִּי יְיָ אֱלֹהֵיכֶם הוּא אֱלֹהֵי הָאֱלֹהִים וַאֲדֹנֵי הָאֲדֹנִים, הָאֵל הַגָּדֹל הַגִּבֹּר וְהַנּוֹרָא, אֲשֶׁר לֹא יִשָּׂא פָנִים וְלֹא יִקַּח שֹֽׁחַד. וּכְתִיב בָּתְרֵהּ: עֹשֶׂה מִשְׁפַּט יָתוֹם וְאַלְמָנָה, וְאֹהֵב גֵּר לָֽתֶת־לוֹ
