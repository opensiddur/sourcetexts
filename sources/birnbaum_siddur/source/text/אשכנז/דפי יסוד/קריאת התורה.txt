קַדִּישׁ עַל אַרְעָא, עוֹבַד גְּבוּרְתֵּהּ; קַדִּישׁ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא יְיָ צְבָאוֹת; מַלְיָא כׇל אַרְעָא זִיו יְקָרֵהּ. וַתִּשָּׂאֵֽנִי רֽוּחַ, וָאֶשְׁמַע אַחֲרַי קוֹל רַֽעַשׁ גָּדוֹל: בָּרוּךְ כְּבוֹד יְיָ מִמְּקוֹמוֹ.<קטע סוף=ואתה קדוש 1/> <קטע התחלה=ואתה קדוש 2/>וּנְטָלַתְנִי רוּחָא, וְשִׁמְעֵת בַּתְרַי קָל זִֽיעַ סַגִּיא דִּי מְשַׁבְּחִין וְאָמְרִין: בְּרִיךְ יְקָרָא דַייָ מֵאֲתַר בֵּית שְׁכִינְתֵּהּ. יְיָ<קטע סוף=ואתה קדוש 2/> <קטע התחלה=ואתה קדוש 3/>יִמְלֺךְ לְעוֹלָם וָעֶד. יְיָ מַלְכוּתֵהּ (קָאֵם) לְעָלַם וּלְעָלְמֵי עָלְמַיָּא. יְיָ אֱלֺהֵי אַבְרָהָם יִצְחָק וְיִשְׂרָאֵל אֲבוֹתֵֽינוּ, שׇׁמְרָה־זֹאת לְעוֹלָם, לְיֵֽצֶר מַחְשְׁבוֹת לְבַב עַמֶּֽךָ, וְהָכֵן לְבָבָם אֵלֶֽיךָ. וְהוּא רַחוּם, יְכַפֵּר עָוֺן וְלֺא יַשְׁחִית, וְהִרְבָּה לְהָשִׁיב אַפּוֹ, וְלֺא יָעִיר כׇּל חֲמָתוֹ. כִּי אַתָּה, אֲדֹנָי, טוֹב וְסַלָּח וְרַב חֶֽסֶד לְכׇל קֹרְאֶֽיךָ.<קטע סוף=ואתה קדוש 3/> <קטע התחלה=ואתה קדוש 4/>צִדְקָתְךָ צֶֽדֶק לְעוֹלָם, וְתוֹרָתְךָ אֱמֶת. תִּתֵּן אֱמֶת לְיַעֲקֹב, חֶֽסֶד לְאַבְרָהָם, אֲשֶׁר נִשְׁבַּֽעְתָּ לַאֲבוֹתֵֽינוּ מִֽימֵי קֶֽדֶם. בָּרוּךְ יְיָ, {{נוסח|יוֹם יוֹם; "רִבּוֹנוֹ שֶׁל עוֹלָם" לשלוש רגלים:

<קטע התחלה=ריבונו של עולם מלא משאלות/>רִבּוֹנוֹ שֶׁל עוֹלָם, מַלֵּא מִשְׁאֲלוֹת לִבִּי לְטוֹבָה, וְהָפֵק רְצוֹנִי וְתֵן שְׁאֵלָתִי, לִי עַבְדְּךָ בֶּן אֲמָתֶֽךָ, וְזַכֵּֽנִי (וְאֶת אִשְׁתִּי וּבָנַי וּבְנוֹתַי) לַעֲשׂוֹת רְצוֹנְךָ בְּלֵבָב שָׁלֵם. וּמַלְּטֵֽנוּ מִיֵּֽצֶר הָרָע, וְתֵן חֶלְקֵֽנוּ בְּתוֹרָתֶֽךָ, וְזַכֵּֽנוּ שֶׁתִּשְׁרֶה שְׁכִינָתְךָ עָלֵֽינוּ, וְהוֹפַע עָלֵֽינוּ רֽוּחַ חׇכְמָה וּבִינָה, רֽוּחַ עֵצָה וּגְבוּרָה, רֽוּחַ דַּֽעַת וְיִרְאַת יְיָ. וְכֵן יְהִי רָצוֹן מִלְּפָנֶֽיךָ, יְיָ אֱלֹהֵֽינוּ וֵאלֹהֵי אֲבוֹתֵֽינוּ, שֶׁתְּזַכֵּֽנוּ לַעֲשׂוֹת מַעֲשִׂים טוֹבִים בְּעֵינֶֽיךָ וְלָלֶֽכֶת בְּדַרְכֵי יְשָׁרִים לְפָנֶֽיךָ. וְקַדְּשֵֽׁנוּ בְּמִצְוֺתֶֽיךָ, כְּדֵי שֶׁנִּזְכֶּה לְחַיִּים טוֹבִים וַאֲרֻכִּים וּלְחַיֵּי הָעוֹלָם הַבָּא; וְתִשְׁמְרֵֽנוּ מִמַּעֲשִׂים רָעִים וּמִשָּׁעוֹת רָעוֹת הַמִּתְרַגְּשׁוֹת לָבֹא לָעוֹלָם. וְהַבּוֹטֵֽחַ בַּייָ חֶֽסֶד יְסוֹבְבֶֽנּוּ. אָמֵן.<קטע סוף=ריבונו של עולם מלא משאלות/>

<קטע התחלה=ריבון העולם לימים הנוראים/>רִבּוֹן הָעוֹלָם, מַלֵּא מִשְׁאֲלוֹתֵֽינוּ לְטוֹבָה, וְהָפֵק רְצוֹנֵֽנוּ וְתֶן־לָֽנוּ שְׁאֵלָתֵֽנוּ, וּמְחַל עַל כׇּל עֲוֺנֹתֵֽינוּ וְעַל כָּל עֲוֺנוֹת אַנְשֵׁי בָתֵּֽינוּ מְחִילָה בְחֶֽסֶד, מְחִילָה בְרַחֲמִים. וְטַהֲרֵֽנוּ מֵחֲטָאֵֽינוּ וּמֵעֲוֺנֹתֵֽינוּ וּמִפְּשָׁעֵֽינוּ; וְזׇכְרֵֽנוּ בְּזִכָּרוֹן טוֹב לְפָנֶֽיךָ, וּפׇקְדֵֽנוּ בִּפְקֻדַּת יְשׁוּעָה וְרַחֲמִים. וְזׇכְרֵֽנוּ לְחַיִּים טוֹבִים וַאֲרֻכִּים וּלְשָׁלוֹם, וּפַרְנָסָה וְכַלְכָּלָה. וְתֶן־לָֽנוּ לֶֽחֶם לֶאֱכוֹל וּבֶֽגֶד לִלְבּוֹשׁ, וְעֹֽשֶׁר וְכָבוֹד וְאֹֽרֶךְ יָמִים, לַהֲגוֹת בְּתוֹרָתֶֽךָ וּלְקַיֵּם מִצְוֺתֶֽיהָ; וְשֵֽׂכֶל וּבִינָה לְהָבִין וּלְהַשְׂכִּיל עִמְקֵי סוֹדוֹתֶֽיהָ. וּשְׁלַח רְפוּאָה לְכׇל מַכְאוֹבֵֽינוּ, וּתְבָרֵךְ אֶת כׇּל מַעֲשֵׂה יָדֵֽינוּ. וְתִגְזוֹר עָלֵֽינוּ גְזֵרוֹת טוֹבוֹת יְשׁוּעוֹת וְנֶחָמוֹת, וּתְבַטֵּל מֵעָלֵֽינוּ כׇּל גְּזֵרוֹת קָשׁוֹת. וְתַטֶּה לֵב הַמַּלְכוּת וְיוֹעֲצֶֽיהָ וְשָׂרֶֽיהָ עָלֵֽינוּ לְטוֹבָה. אָמֵן, וְכֵן יְהִי רָצוֹן.<קטע סוף=ריבון העולם לימים הנוראים/>

<קטע התחלה=הוראה לפני יקום פורקן הראשון/>התפילות "יְקוּם פֻּרְקָן" (הראשון והשני) ו"מִי שֶׁבֵּרַךְ אֲבוֹתֵֽינוּ" נאמרות בשבת בלבד ולא ביום טוב.{{ש}}בשבת הש"ץ לתפילת מוסף מתחיל כאן ב"יְקוּם פֻּרְקָן", וביום טוב הוא מתחיל להלן ב[[#תפילה לשלום המדינה|"אָבִֽינוּ שֶׁבַּשָּׁמַֽיִם צוּר יִשְׂרָאֵל וְגוֹאֲלוֹ"]] (בארץ) או ב[[#תפילה בשלומה של מלכות|"הַנּוֹתֵן תְּשׁוּעָה לַמְּלָכִים"]] (בחו"ל).<קטע סוף=הוראה לפני יקום פורקן הראשון/>

<קטע התחלה=יקום פורקן הראשון מילים/>יְקוּם פֻּרְקָן מִן שְׁמַיָּא, חִנָּא וְחִסְדָּא וְרַחֲמֵי, וְחַיֵּי אֲרִיכֵי וּמְזוֹנֵי רְוִיחֵי {{נוסח|וְסַיַּעְתָּא|=בירנבוים ונוסח אשכנז הוותיק|בסידור "בית תפילה" לר' זלמן הענא תיקן ל"וְסִיַּעְתָּא" (בחיריק)}} דִשְׁמַיָּא, וּבַרְיוּת גּוּפָא וּנְהוֹרָא מַעַלְיָא, זַרְעָא חַיָּא וְקַיָּמָא, זַרְעָא דִּי לָא {{נוסח|יִפְסֻק|=בירנבוים|יִפְסוֹק}} וְדִי לָא {{נוסח|יִבְטֻל|=בירנבוים|יִבְטוֹל}} מִפִּתְגָּמֵי אוֹרַיְתָא, לְמָרָנָן וְרַבָּנָן, חֲבוּרָתָא קַדִּישָׁתָא דִּי {{נוסח|בְּאַרְעָא|=בירנבוים|בְאַרְעָא}} דְיִשְׂרָאֵל וְדִי בְּבָבֶל; לְרֵישֵׁי כַלֵּי וּלְרֵישֵׁי גַלְוָתָא, וּלְרֵישֵׁי מְתִיבָתָא וּלְדַיָּנֵי {{נוסח|דִי בָבָא|=בירנבוים בעקבות היידנהיים|נוסח אשכנז הוותיק=דְבָבָא (תיבה אחת)}}; לְכׇל תַּלְמִידֵיהוֹן וּלְכׇל תַּלְמִידֵי תַלְמִידֵיהוֹן, וּלְכׇל {{נוסח|מָן|=בירנבוים|מָאן}} דְּעָסְקִין בְּאוֹרַיְתָא. מַלְכָּא דְעָלְמָא יְבָרֵךְ יַתְהוֹן, {{נוסח|יְפִישׁ|=בירנבוים|יַפִּישׁ}} חַיֵּיהוֹן וְיַסְגֵּא יוֹמֵיהוֹן וְיִתֵּן {{נוסח|אַרְכָא|=בירנבוים|אַרְכָּא}} לִשְׁנֵיהוֹן, וְיִתְפָּרְקוּן וְיִשְׁתֵּזְבוּן {{נוסח|מִן כׇּל|=בירנבוים בעקבות תיקונו של ר' זלמן הענא|מִכׇּל (וכן "ומכל מרעין בישין"}} עָקָא וּמִן כׇּל מַרְעִין בִּישִׁין. מָרָן דִּי בִשְׁמַיָּא יְהֵא בְסַעְדְּהוֹן כׇּל זְמַן וְעִדָּן, וְנֹאמַר אָמֵן.<קטע סוף=יקום פורקן הראשון מילים/>

[<קטע התחלה=ציון של תרגום/>תַּרְגּוּם:<קטע סוף=ציון של תרגום/> <קטע התחלה=תרגום ליקום פורקן הראשון/>תַּעֲמֹד יְשׁוּעָה מִן הַשָּׁמַֽיִם, חֵן וָחֶֽסֶד וְרַחֲמִים, וְחַיִּים אֲרֻכִּים וּמָזוֹן בְּרֶֽוַח וְעֶזְרַת הַשָּׁמַֽיִם, וּבְרִיאוּת הַגּוּף וְאוֹר (עֵינַֽיִם) מְעֻלֶּה, זֶֽרַע חַי וְקַיָּם, זֶֽרַע אֲשֶׁר לֹא יִפְסֹק וַאֲשֶׁר לֹא יִבְטֹל מִדִּבְרֵי תּוֹרָה, לְמוֹרֵֽינוּ וְרַבּוֹתֵֽינוּ, הַחֲבוּרָה הַקְּדוֹשָׁה אֲשֶׁר בְּאֶֽרֶץ יִשְׂרָאֵל וַאֲשֶׁר בְּבָבֶל; לְרָאשֵׁי כַלָּה וּלְרָאשֵׁי הַגּוֹלָה, וּלְרָאשֵׁי הַיְּשִׁיבוֹת וְלַשּׁוֹפְטִים בַּשַּֽׁעַר; לְכׇל תַּלְמִידֵיהֶם וּלְכׇל תַּלְמִידֵי תַּלְמִידֵיהֶם, וּלְכׇל מִי שֶׁעוֹסְקִים בַּתּוֹרָה. מֶֽלֶךְ הָעוֹלָם יְבָרֵךְ אוֹתָם, יַרְבֶּה חַיֵּיהֶם וְיוֹסִיף עַל יְמֵיהֶם וְיַאֲרִיךְ שְׁנוֹתֵיהֶם, וְיִוָּשְׁעוּ וְיִנָּצְלוּ מִכׇּל צָרָה וּמִכׇּל חֳלָיִים רָעִים. אֲדוֹנֵֽנוּ שֶׁבַּשָּׁמַֽיִם יִהְיֶה לְמִשְׁעָן לָהֶם בְּכׇל עֵת וְשָׁעָה, וְנֹאמַר אָמֵן.<קטע סוף=תרגום ליקום פורקן הראשון/>]

<קטע התחלה=הוראה לפני יקום פורקן השני/>יחיד אינו אומר את תפילת "יְקוּם פֻּרְקָן" השני ואת "מִי שֶׁבֵּרַךְ אֲבוֹתֵֽינוּ".<קטע סוף=הוראה לפני יקום פורקן השני/>

<קטע התחלה=יקום פורקן השני מילים/>יְקוּם פֻּרְקָן מִן שְׁמַיָּא, חִנָּא וְחִסְדָּא וְרַחֲמֵי, וְחַיֵּי אֲרִיכֵי וּמְזוֹנֵי רְוִיחֵי {{נוסח|וְסַיַּעְתָּא|=בירנבוים|וְסִיַּעְתָּא}} דִשְׁמַיָּא, וּבַרְיוּת גּוּפָא וּנְהוֹרָא מַעַלְיָא, זַרְעָא חַיָּא וְקַיָּמָא, זַרְעָא דִּי לָא {{נוסח|יִפְסֻק|=בירנבוים|יִפְסוֹק}} וְדִי לָא {{נוסח|יִבְטֻל|=בירנבוים|יִבְטוֹל}} מִפִּתְגָּמֵי אוֹרַיְתָא, לְכׇל קְהָלָא קַדִּישָׁא הָדֵן, רַבְרְבַיָּא עִם זְעֵרַיָּא, טַפְלָא וּנְשַׁיָּא. מַלְכָּא דְעָלְמָא יְבָרֵךְ יַתְכוֹן, {{נוסח|יְפִישׁ|=בירנבוים|יַפִּישׁ}} חַיֵּיכוֹן וְיַסְגֵּא יוֹמֵיכוֹן וְיִתֵּן {{נוסח|אַרְכָא|=בירנבוים|אַרְכָּא}} לִשְׁנֵיכוֹן, וְתִתְפָּרְקוּן וְתִשְׁתֵּיזְבוּן מִן כׇּל עָקָא וּמִן כׇּל מַרְעִין בִּישִׁין. מָרָן דִּי בִשְׁמַיָא יְהֵא בְסַעְדְּכוֹן כׇּל זְמַן וְעִדָּן, וְנֹאמַר אָמֵן.<קטע סוף=יקום פורקן השני מילים/>

[{{#קטע:הסידור השלם (בירנבוים)/אשכנז/דפי יסוד/קריאת התורה|ציון של תרגום}} <קטע התחלה=תרגום ליקום פורקן השני/>תַּעֲמֹד יְשׁוּעָה מִן הַשָּׁמַֽיִם, חֵן וָחֶֽסֶד וְרַחֲמִים, וְחַיִּים אֲרֻכִּים וּמָזוֹן בְּרֶֽוַח וְעֶזְרַת הַשָּׁמַיִם, וּבְרִיאוּת הַגּוּף וְאוֹר (עֵינַֽיִם) מְעֻלֶּה, זֶֽרַע חַי וְקַיָּם, זֶֽרַע אֲשֶׁר לֹא יִפְסֹק וַאֲשֶׁר לֹא יִבְטֹל מִדִּבְרֵי תּוֹרָה, לְכׇל הַקָּהָל הַקָּדוֹשׁ הַזֶּה, הַגְּדוֹלִים עִם הַקְּטַנִּים, הַטַּף וְהַנָּשִׁים. מֶֽלֶךְ הָעוֹלָם יְבָרֵךְ אֶתְכֶם, יַרְבֶּה חַיֵּיכֶם וְיוֹסִיף עַל יְמֵיכֶם וְיַאֲרִיךְ שְׁנוֹתֵיכֶם, וְתִוָּשְׁעוּ וְתִנָּצְלוּ מִכׇּל צָרָה וּמִכׇּל חֳלָיִים רָעִים. אֲדוֹנֵֽנוּ שֶׁבַּשָּׁמַֽיִם יִהְיֶה לְמִשְׁעָן לָכֶם בְּכׇל עֵת וְשָׁעָה, וְנֹאמַר אָמֵן.<קטע סוף=תרגום ליקום פורקן השני/>]

<קטע התחלה=כותרת לתפילה בשלומה של מלכות בארצות הברית/>{{עוגן בשורה|תפילה בשלומה של מלכות}}תְּפִלָּה בִשְׁלוֹמָהּ שֶׁל מַלְכוּת [בְּאַרְצוֹת הַבְּרִית]<קטע סוף=כותרת לתפילה בשלומה של מלכות בארצות הברית/>

<קטע התחלה=הוראה כללית לפני הנותן תשועה/>בחוץ לארץ, לפני התפילה לשלום מדינת ישראל, יש להתפלל לשלומה של המלכות במקום<קטע סוף=הוראה כללית לפני הנותן תשועה/>; <קטע התחלה=הוראה לפני הנותן תשועה בארה"ב/>להלן מובא נוסח המתאים לארצות הברית.<קטע סוף=הוראה לפני הנותן תשועה בארה"ב/><קטע התחלה=הערה על התפילה בשלומה של מלכות בארצות הברית/>{{הערה|ב'''סידור עם ישראל''' בעריכתו של פלטיאל בירנבוים (ניו יורק, 1979), הדפיס בירנבוים את "תְּפִלָּה בִשְׁלוֹמָהּ שֶׁל מַלְכוּת" בנוסח סתמי: "וִינַשֵּׂא לְמַֽעְלָה אֶת כׇּל שָׂרֵי הָאָֽרֶץ הַזֹּאת" (עמ' 200), כנראה כדי שהתפילה תהיה מתאימה בעוד מדינות ולא רק בארצות הברית. אך אין בניסוח הזה מענה מלא לצרכיהם של המתפללים בארצות אחרות. לפרטים נוספים על התפילה, והפניות למקורות מידע עליה, ראו במאמר ב'''[[:W:תפילה לשלום המלכות|וויקיפדיה]]'''.}}<קטע סוף=הערה על התפילה בשלומה של מלכות בארצות הברית/>

<קטע התחלה=מאת הרבנים הראשיים שבארץ ישראל/>{{נוסח|מאת הרבנים הראשיים שבארץ ישראל|מֵאֵת הָרַבָּנִים הָרָאשִׁיִּים שֶׁבְּאֶֽרֶץ יִשְׂרָאֵל}}<קטע סוף=מאת הרבנים הראשיים שבארץ ישראל/><קטע התחלה=הערה על התפילה לשלום המדינה/>{{הערה|התפילה לשלום המדינה חוברה ע"י [[:W:יצחק אייזיק הלוי הרצוג|הרב יצחק אייזיק הלוי הרצוג]] ונעשו בה תיקונים קלים ע"י הסופר [[:W:שמואל יוסף עגנון|ש"י עגנון]]; היא נתקנה ע"י שני הרבנים הראשיים, הרב הרצוג ו[[:W:בן-ציון מאיר חי עוזיאל|הרב בן ציון מאיר חי עוזיאל]], במהלך [[:W:מלחמת העצמאות|מלחמת העצמאות]] (כ-4 חודשים אחרי [[:W:הכרזת העצמאות|הכרזת העצמאות]]). לפרטים נוספים על התפילה, והפניות למקורות מידע עליה, ראו במאמר ב'''[[:W:תפילה לשלום המדינה|וויקיפדיה]]'''. הוספנו את התפילה כאן, לפני ברכת החודש, ולאחריה את התפילה לשלום חיילי צה"ל, בהשראתו של '''סידור עם ישראל''' (בעריכתו של בירנבוים). בסידורו המקורי (ניו יורק, 1949) הדפיס בירנבוים את התפילה לשלום מדינת ישראל בעמודים האחרונים של הסידור (עמ' [[עמוד:Philip Birnbaum - ha-Siddur ha-Shalem (The Daily Prayer Book,1949).pdf/814|789]]-[[עמוד:Philip Birnbaum - ha-Siddur ha-Shalem (The Daily Prayer Book,1949).pdf/815|790]] בעברית ובאנגלית במקביל). במהדורות הבאות של סידוריו הוא הוסיף גם את השירים [[:W:מעל פסגת הר הצופים|עַל פִּסְגַּת הַר הַצּוֹפִים]] ו[[:W:התקווה|הַתִּקְוָה]]. ואילו ב'''סידור עם ישראל''' הכניס בירנבוים את "תְּפִלָּה לִשְׁלוֹם מְדִינַת יִשְׂרָאֵל" ואת "מִי שֶׁבֵּרַךְ לְצַהַ"ל" בפנים הסידור, בתוך שחרית לשבת ויום טוב (עמ' 200-201). לפניהם השאיר בירנבוים את ה"תְּפִלָּה בִשְׁלוֹמָהּ שֶׁל מַלְכוּת", אך הפעם הדפיס אותה בנוסח סתמי (ראו הערה על "תפילה בשלומה של מלכות" [[#תפילה בשלומה של מלכות|לעיל]]).}}<קטע סוף=הערה על התפילה לשלום המדינה/>

<קטע התחלה=מאת הרב הצבאי הראשי/>{{נוסח|מאת הרב הצבאי הראשי (שנת ה' תשי"ז).|מֵאֵת הָרַב הַצְּבָאִי הָרָאשִׁי}}<קטע סוף=מאת הרב הצבאי הראשי/><קטע התחלה=הערה על התפילה לשלום חיילי צה"ל/>{{הערה|תפילה זו תוקנה על ידי [[:W:שלמה גורן|הרב שלמה גורן]], הרב הראשי לצה"ל, בעיצומו של [[:W:מלחמת סיני|מבצע קדש]] (מלחמת סיני). לפרטים נוספים על התפילה, והפניות למקורות מידע עליה, ראו במאמר ב'''[[:W:תפילה לשלום חיילי צה"ל|וויקיפדיה]]'''. התפילה מובאת כאן לפי גרסתו המקורית שחיבר ופרסם הרב גורן, עם שתי תוספות: "וְאַנְשֵׁי כֹּחוֹת הַבִּטָּחוֹן", "וּבְכָל מָקוֹם שֶׁהֵם"; לפרטים על הנוסח המקורי ועל התוספות ראו במאמרו של יואל רפל, [https://musaf-shabbat.com/2014/09/12/%D7%91%D7%9B%D7%9C-%D7%9E%D7%A7%D7%95%D7%9D-%D7%A9%D7%94%D7%9D-%D7%99%D7%95%D7%90%D7%9C-%D7%A8%D7%A4%D7%9C/ "בכל מקום שהם"], שהופיע במוסף "שבת", '''מקור ראשון''', י"ז אלול תשע"ד; ובמיוחד שימו לב לפסקה השנייה לפני סוף המאמר. לדוגמה של סידור  מתקופתו של הרב גורן כרב הראשי לצה"ל, שבו נדפסה התפילה שלו לשלום חיילי צה"ל, ראו '''[https://www.nli.org.il/he/books/NNL_ALEPH990022607210205171/NLI סידור מאירת עיניים]''': כולל כל התפלות לכל ימות השנה ותפלות מיוחדות לצבא הגנה לישראל וליום העצמאות, נוסח ספרד, הוצאה מיוחדת לבתי הכנסת הצבאיים ע"י הרבנות הצבאית הראשית ט"ו למדינת ישראל התשכ"ג (תל־אביב: סיני, תשכ"ג), עמ' 573 ("מי שברך לחיילי צה"ל").}}<קטע סוף=הערה על התפילה לשלום חיילי צה"ל/>

<קטע התחלה=הוא יברך את חיילי צה"ל/>מִי שֶׁבֵּרַךְ אֲבוֹתֵֽינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב, הוּא יְבָרֵךְ אֶת חַיָּלֵי צְבָא הַהֲגַנָּה לְיִשְׂרָאֵל וְאַנְשֵׁי כֹּחוֹת הַבִּטָּחוֹן, הָעוֹמְדִים עַל מִשְׁמַר אַרְצֵֽנוּ וְעָרֵי אֱלֹהֵֽינוּ, מִגְּבוּל הַלְּבָנוֹן וְעַד מִדְבַּר מִצְרַיִם, וּמִן הַיָּם הַגָּדוֹל עַד לְבוֹא הָעֲרָבָה וּבְכׇל מָקוֹם שֶׁהֵם, בַּיַּבָּשָׁה בָּאֲוִיר וּבַיָּם.<קטע סוף=הוא יברך את חיילי צה"ל/>

<קטע התחלה=יתן ה' את אויבנו/>יִתֵּן יְיָ אֶת אוֹיְבֵֽינוּ הַקָּמִים עָלֵֽינוּ נִגָּפִים לִפְנֵיהֶם! הַקָּדוֹשׁ בָּרוּךְ הוּא {{נוסח|יִשְׁמְרֵם וְיַצִּילֵם|=הנוסח המקורי שכתב הרב גורן; עוד נוסחאות נפוצות: "יִשְׁמֹר וְיַצִּיל אֶת חַיָּלֵינוּ" (שאינו מכיל את "אַנְשֵׁי כֹּחוֹת הַבִּטָּחוֹן") או  "יִשְׁמֹר וְיַצִּיל אֶת מְגִנֵּי אֶֽרֶץ קָדְשֵֽׁנוּ" (שמכיל את כולם).}} מִכׇּל צָרָה וְצוּקָה, וּמִכׇּל נֶֽגַע וּמַחֲלָה, וְיִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכׇל מַעֲשֵׂה יְדֵיהֶם. יַדְבֵּר שׂוֹנְאֵֽינוּ תַּחְתֵּיהֶם, וִיעַטְּרֵם בְּכֶֽתֶר יְשׁוּעָה וּבַעֲטֶֽרֶת נִצָּחוֹן. וִיקֻיַּם בָּהֶם הַכָּתוּב: כִּי יְיָ אֱלֹהֵיכֶם הַהֹלֵךְ עִמָּכֶם, לְהִלָּחֵם לָכֶם עִם אֹיְבֵיכֶם לְהוֹשִֽׁיעַ אֶתְכֶם. וְנֹאמַר: אָמֵן.<קטע סוף=יתן ה' את אויבנו/>

<קטע התחלה=נכתבה על ידי הרב שלמה גורן/>{{נוסח|נכתבה על־ידי הרב שלמה גורן בעקבות מלחמת יום הכיפורים, ונערכה ע"י בנו רמי גורן לאחר שמחת תורה תשפ"ד.|מֵאֵת הָרַב הָרָאשִׁי לְיִשְׂרָאֵל}}<קטע סוף=נכתבה על ידי הרב שלמה גורן/><קטע התחלה=הערה על התפילה לשבויים/>{{הערה|תפילה זו נכתבה על־ידי [[:W:שלמה גורן|הרב שלמה גורן]], הרב הראשי לישראל ולפני כן הרב הראשי לצה"ל, בעקבות [[:W:מלחמת יום הכיפורים|מלחמת יום הכיפורים]], ועודכנה על־ידי בנו רמי גורן לאחר [[:W:טבח שבעה באוקטובר|שמחת תורה תשפ"ד]]. למידע על התפילה ועל העדכון ראו סיון רהב-מאיר, [https://www.inn.co.il/news/617173 "התגלתה תפילה לשלום השבויים"] (באתר '''ערוץ 7''' ג' בחשוון תשפ"ד 18.10.23); התפילה פורסמה לראשונה ב[https://sivanrahavmeir.com/%D7%94%D7%97%D7%9C%D7%A7-%D7%94%D7%99%D7%95%D7%9E%D7%99/%D7%A9%D7%9E%D7%A2%D7%AA%D7%9D-%D7%A2%D7%9C-%D7%94%D7%AA%D7%A4%D7%99%D7%9C%D7%94-%D7%9C%D7%A9%D7%9C%D7%95%D7%9D-%D7%94%D7%A9%D7%91%D7%95%D7%99%D7%99%D7%9D/ אתר האישי שלה] יום קודם. הוספנו את התפילה כאן בהשראתו של '''סידור קורן''' החדש (מהדורת פומרנץ ירושלים תשע"ט, עמ' 247), אמנם שם היא מופיעה בגרסה אחרת.}}<קטע סוף=הערה על התפילה לשבויים/>

<קטע התחלה=תפילה לשבויים 1/>מִי שֶׁבֵּרַךְ אֲבוֹתֵֽינוּ, אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה אַהֲרֹן דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ וְיִשְׁמֹר וְיַצִּיל אֶת בָּנֵינוּ הַיְּקָרִים, חַיָּלֵי צְבָא הַהֲגָנָה לְיִשְׂרָאֵל וּשְׁאָר הַלּוֹחֲמִים לְמַֽעַן בִּטְחוֹן הָעָם וְהָאָֽרֶץ, וְאֶזְרָחִים – זְקֵנִים וּצְעִירִים, נָשִׁים, גְּבָרִים וְטַף – הַנִּמְצָאִים בַּצָּרָה וּבִמְצוּקָה, וּנְתוּנִים לִמְשִׁסָּה וְסַכָּנָה בַּשֶּֽׁבִי.<קטע סוף=תפילה לשבויים 1/>

<קטע התחלה=תפילה לשבויים 2/>הַקָּדוֹשׁ בָּרוּךְ הוּא יִמָּלֵא רַחֲמִים עֲלֵיהֶם לְקַיְּמָם וּלְחַזְּקָם וּלְהַחֲיוֹת רוּחָם וְתִקְוָתָם, לְהַצִּיל מִמַּסְגֵּר נַפְשָׁם, לְהוֹצִיאָם מִצָּרָה לִרְוָחָה וּמִשִּׁעְבּוּד לִגְאֻלָּה.<קטע סוף=תפילה לשבויים 2/>

<קטע התחלה=הוראה לזכרון קדושים/>לזכרון קדושים:<קטע סוף=הוראה לזכרון קדושים/>

<קטע התחלה=יזכור לקדושים/>יִזְכּוֹר אֱלֹהִים נִשְׁמוֹת כׇּל הַקְּדוֹשִׁים וְהַטְּהוֹרִים שֶׁנֶּהֶרְגוּ, שֶׁנִּשְׁחֲטוּ וְשֶׁנִּשְׂרְפוּ, וְשֶׁנִּטְבְּעוּ וְשֶׁנֶּחְנְקוּ עַל קִדּוּשׁ הַשֵּׁם. בַּעֲבוּר שֶׁנּוֹדְרִים צְדָקָה בְּעַד הַזְכָּרַת נִשְׁמוֹתֵיהֶם, בִּשְׂכַר זֶה, תִּהְיֶֽינָה נַפְשׁוֹתֵיהֶם צְרוּרוֹת בִּצְרוֹר הַחַיִּים עִם נִשְׁמוֹת אַבְרָהָם יִצְחָק וְיַעֲקֹב, שָׂרָה רִבְקָה רָחֵל וְלֵאָה, וְעִם שְׁאָר צַדִּיקִים וְצִדְקָנִיּוֹת שֶׁבְּגַן־עֵֽדֶן, וְנֹאמַר אָמֵן.<קטע סוף=יזכור לקדושים/>

<קטע התחלה=הוראה לזכרון איש זכר/>אזכרה לזכר ("אֵל מָלֵא רַחֲמִים"):<קטע סוף=הוראה לזכרון איש זכר/>

<קטע התחלה=יזכור לאיש זכר/>אֵל מָלֵא רַחֲמִים, שׁוֹכֵן בַּמְּרוֹמִים, הַמְצֵא מְנוּחָה נְכוֹנָה תַּֽחַת כַּנְפֵי הַשְּׁכִינָה, בְּמַעֲלוֹת קְדוֹשִׁים וּטְהוֹרִים כְּזֹֽהַר הָרָקִֽיעַ מַזְהִירִים, אֶת נִשְׁמַת (פלוני בן פלוני) שֶׁהָלַךְ לְעוֹלָמוֹ. בַּעֲבוּר שֶׁנָּדְרוּ צְדָקָה בְּעַד הַזְכָּרַת נִשְׁמָתוֹ, בְּגַן־עֵֽדֶן תְּהֵא מְנוּחָתוֹ. לָכֵן בַּֽעַל הָרַחֲמִים יַסְתִּירֵֽהוּ בְּסֵֽתֶר כְּנָפָיו לְעוֹלָמִים, וְיִצְרוֹר בִּצְרוֹר הַחַיִּים אֶת נִשְׁמָתוֹ. יְיָ הוּא נַחֲלָתוֹ; וְיָנֽוּחַ עַל מִשְׁכָּבוֹ בְּשָׁלוֹם, וְנֹאמַר אָמֵן.<קטע סוף=יזכור לאיש זכר/>

<קטע התחלה=הוראה לזכרון נקבה/>אזכרה לנקבה ("אֵל מָלֵא רַחֲמִים"):<קטע סוף=הוראה לזכרון נקבה/>

<קטע התחלה=יזכור לנקבה/>אֵל מָלֵא רַחֲמִים, שׁוֹכֵן בַּמְּרוֹמִים, הַמְצֵא מְנוּחָה נְכוֹנָה תַּֽחַת כַּנְפֵי הַשְּׁכִינָה, בְּמַעֲלוֹת קְדוֹשִׁים וּטְהוֹרִים כְּזֹֽהַר הָרָקִֽיעַ מַזְהִירִים, אֶת נִשְׁמַת (פלונית בת פלוני) שֶׁהָלְכָה לְעוֹלָמָהּ. בַּעֲבוּר שֶׁנָּדְרוּ צְדָקָה בְּעַד הַזְכָּרַת נִשְׁמָתָהּ, בְּגַן־עֵֽדֶן תְּהֵא מְנוּחָתָהּ. לָכֵן בַּֽעַל הָרַחֲמִים יַסְתִּירֶֽהָ בְּסֵֽתֶר כְּנָפָיו לְעוֹלָמִים, וְיִצְרוֹר בִּצְרוֹר הַחַיִּים אֶת נִשְׁמָתָהּ. יְיָ הוּא נַחֲלָתָהּ; וְתָנֽוּחַ עַל מִשְׁכָּבָהּ בְּשָׁלוֹם, וְנֹאמַר אָמֵן.<קטע סוף=יזכור לנקבה/>

א. <קטע התחלה=הוראה לזכרון קדושי השואה יד ושם ארוך/>לזכרון קדושי השואה (על בסיס הנוסח הארוך של יד ושם):{{הערה|מבוסס על הנוסח הרשמי של [https://www.yadvashem.org/he/education/educational-materials/ceremonies/tehilim.html יד ושם], אך נעשו בו תיקונים קלים לפי סידורי תפילה אחדים. לגבי "בַּגּוֹלָה" במקום "בְּגוֹלַת אֵירֽוֹפָּה" ראו '''[https://livluv.org.il/%D7%A8%D7%92%D7%A2-%D7%94%D7%99%D7%A1%D7%98%D7%95%D7%A8%D7%99-%D7%A9%D7%9C-%D7%A2%D7%A9%D7%99%D7%99%D7%AA-%D7%A6%D7%93%D7%A7-%D7%AA%D7%9C%D7%9E%D7%99%D7%93%D7%AA-%D7%94%D7%AA%D7%99%D7%9B%D7%95%D7%9F/ כאן]'''.}}<קטע סוף=הוראה לזכרון קדושי השואה יד ושם ארוך/>

<קטע התחלה=יזכור לקדושי השואה יד ושם ארוך/>יִזְכּוֹר אֱלֹהִים אֶת כׇּל הַנְּשָׁמוֹת שֶׁל {{נוסח|קְהִלּוֹת|=מ2,פ,ר3|י=כׇּל קְהִלּוֹת}} בֵּית יִשְׂרָאֵל {{נוסח|בַּגּוֹלָה|=י|מ,פ,ר3=בְּגוֹלַת אֵירֽוֹפָּה}}, {{נוסח|שֶׁהֹעֲלוּ|=מ2,ר3 (ע"פ נחום ב,ח; דה"ב כ,לד; שופטים ו,כח)|פ=שֶׁהוֹעֲלוּ|י=שֶׁהוּעֲלוּ}} עַל מוֹקֵד בִּשְׁנוֹת הַשּׁוֹאָה תַּ"שׁ–תַּשָׁ"ה: שִׁשָּׁה מִלְיוֹן אֲנָשִׁים וְנָשִׁים, יְלָדִים וִילָדוֹת, בַּחוּרִים וּבְתוּלוֹת, זָקֵן וָטַף, שֶׁנֶּהֶרְגוּ וְשֶׁנִּטְבְּחוּ בְּאַכְזָרִיּוּת אֲיֻמָּה, וְנִרְצְחוּ בְּרֶֽצַח הֲמוֹנִים בִּמְקוֹמוֹת מְגוּרֵיהֶם, בֶּעָרִים, בָּעֲיָרוֹת וּבַכְּפָרִים; וְיִתְרָם {{נוסח|הוּבְלוּ כַּצֹּאן לַטֶּֽבַח|=מ2,פ,ר3 אך הניקוד לפי ישעיהו מג,ז (במקום "לָטֶֽבַח")|י=הוּבְלוּ}} לְמַחֲנוֹת הָרִכּוּז וְנִסְפּוּ בְּמִיתוֹת מְשֻׁנּוֹת וְנוֹרָאוֹת, וְנִשְׂרְפוּ לְאֵֽפֶר בְּכִבְשְׁנֵי־הָאֵשׁ שֶׁל מַחֲנוֹת הַהַשְׁמָדָה הָאֲיֻמִּים {{נוסח|בְּגֶרְמַֽנְיָה, בְּפוֹלִין|=מ2,פ,ר3|י=בְּגֶרְמַֽנְיָה וּבְפּוֹלִין}} וּבִשְׁאָר אֲרָצוֹת, בִּידֵי עַם הַמְּרַצְּחִים הַגֶּרְמָנִי עִם עוֹזְרָיו הָרוֹצְחִים מִשְּׁאָר הָעַמִּים, שֶׁהָיוּ בְּעֵצָה אַחַת לְהַשְׁמִיד וְלַהֲרֹג וּלְאַבֵּד אֶת הָעָם הַיְּהוּדִי, וְלִמְחוֹת אֶת זֵֽכֶר הַיַּהֲדוּת, וּלְכַלּוֹת כׇּל אֲשֶׁר בְּשֵׁם יִשְׂרָאֵל יִקָּרֵא.

אֵל נְקָמוֹת, שׁוֹפֵט הָאָֽרֶץ, זְכׇר־נָא אֶת נַחֲלֵי הַדָּם שֶׁנִּשְׁפַּךְ כַּמַּֽיִם: דְּמֵי אָבוֹת וּבָנִים, אִמָּהוֹת {{נוסח|וְעוֹלְלֵיהֶן|=י (ע"פ ישעיהו יג,טז אך בסיומת בלשון נקבה)|פ,ר3=וְעוֹלְלֵיהֶם|מ2=וְעוֹלָלֵיהֶם}}, רַבָּנָן וְתַלְמִידֵיהֶם; וְהָשֵׁב גְּמוּל שִׁבְעָתַֽיִם לְצוֹרְרֵי עַמֶּֽךָ. שַׁוְעַת שְׁמַע יִשְׂרָאֵל שֶׁזָּעֲקוּ הַלְּקוּחִים לַמָּֽוֶת, אַל תִּדֹּם; וְאֶנְקַת הַמְּעֻנִּים תַּעֲלֶה לִפְנֵי כִסֵּא כְבוֹדֶֽךָ, לִנְקֹם, בִּמְהֵרָה בְּיָמֵֽינוּ לְעֵינֵֽינוּ, אֶת נִקְמַת דַּם בָּנֶֽיךָ וּבְנוֹתֶֽיךָ הַקְּדוֹשִׁים וְהַטְּהוֹרִים שֶׁלֹּא {{נוסח|זָכוּ לָבוֹא|=מ2,פ,ר3|י=זָכוּ}} לְקֶֽבֶר יִשְׂרָאֵל. כַּכָּתוּב: כִּי דַּם עֲבָדָיו יִקּוֹם, וְנָקָם יָשִׁיב לְצָרָיו, וְכִפֵּר אַדְמָתוֹ עַמּוֹ. {{נוסח|אָמֵן סֶֽלָה.|=מ2,פ,ר3|י=חסר}}<קטע סוף=יזכור לקדושי השואה יד ושם ארוך/>

ב. <קטע התחלה=הוראה לזכרון קדושי השואה יד ושם קצר/>לזכרון קדושי השואה (על בסיס הנוסח הקצר של יד ושם):{{הערה|מבוסס על הנוסח הרשמי של [https://www.yadvashem.org/he/education/educational-materials/ceremonies/tehilim.html יד ושם].}}<קטע סוף=הוראה לזכרון קדושי השואה יד ושם קצר/>

<קטע התחלה=יזכור לקדושי השואה יד ושם קצר/>יִזְכֹּר אֱלֹהִים אֶת נִשְׁמוֹת אֲחֵֽינוּ בְּנֵי יִשְׂרָאֵל, חַלְלֵי הַשּׁוֹאָה וְגִבּוֹרֶֽיהָ, נִשְׁמוֹת שֵׁש־מֵאוֹת רִבְבוֹת אַלְפֵי יִשְׂרָאֵל, שֶׁהוּמְתוּ וְשֶׁנֶּהֶרְגוּ וְשֶׁנֶּחְנְקוּ וְשֶׁנִּקְבְּרוּ חַיִּים, וְאֶת קְהִלּוֹת הַקֹּֽדֶשׁ שֶׁנֶּחְרְבוּ עַל קְדֻשַּׁת הַשֵּׁם.

יִזְכֹּר אֱלֹהִים אֶת עֲקֵדָתָם, עִם עֲקֵדַת שְאַר קְדוֹשֵׁי יִשְׂרָאֵל וְגִבּוֹרָיו מִימֵי עוֹלָם, וְיִצְרֹר בִּצְרוֹר הַחַיִּים אֶת נִשְׁמָתָם. הַנֶּאֱהָבִים וְהַנְּעִימִים בְּחַיֵּיהֶם וּבְמוֹתָם לֹא נִפְרָֽדוּ. יָנֽוּחוּ בְּשָׁלוֹם עַל מִשְׁכְּבוֹתָם, וְנֹאמַר אָמֵן.<קטע סוף=יזכור לקדושי השואה יד ושם קצר/>

<קטע התחלה=יזכור לחללי צה"ל מילים/>יִזְכֹּר אֱלֹהִים אֶת {{נוסח|נִשְמוֹת בָּנָיו־גִּבּוֹרָיו,|=מ4,ר4|מ2,פ,ר3=נִשְמוֹת}} חַיָּלֵי צְבָא {{נוסח|הֲגַנָּה|=מ4,פ,ר3,ר4|מ2=הַהֲגַנָּה}} לְיִשְׂרָאֵל {{נוסח|וְלוֹחֲמֵי הַמַּחְתֶּֽרֶת,|מ,פ2,ר3=וְנִשְׁמוֹת לוֹחֲמֵי הַמַּחְתֶּֽרֶת,|ר4=שֶׁנָּפְלוּ בְּמִלְחֲמוֹת יִשְׂרָאֵל, בִּפְעֻלּוֹת הֲגַנָּה, תַּגְמוּל וּבִטָּחוֹן וּבְעֵת מִלּוּי תַּפְקִידָם, וְנִשְׁמוֹת כׇּל|מ4=חסר עד "שמסרו נפשם")}} {{נוסח|שֶׁמָּסְרוּ נַפְשָׁם|=מ2,מ4,פ,ר3|ר4=וַחֲטִיבוֹת הַלּוֹחֲמִים בְּמַעַרְכוֹת הָעָם, שֶׁחֵרְפוּ נַפְשָׁם לָמוּת}} עַל קְדֻשַּׁת הַשֵּׁם {{נוסח|הָעָם וְהָאָֽרֶץ; וְנָפְלוּ בְּמִלְחֲמוֹת יִשְׂרָאֵל לְשִׁחְרוּר אַרְצֵֽנוּ הַקְּדוֹשָׁה|=מ2,פ,ר3|ר4=חסר|מ4=הָעָם וְהָאָֽרֶץ, וְעָלוּ בְּלַֽהַב תְּקוּמַת יִשְׂרָאֵל בְּמִלְחֶֽמֶת הַשִּׁחְרוּר, בְּמַעַרְכוֹת סִינַי וּבִפְעֻלּוֹת הֲגַנָּה (ואז חסר עד "מנשרים קלו")}}, וּבְעֶזְרַת {{נוסח|יְיָ צְבָאוֹת|=פ,ר3|ר4=אֱלֹהֵי מַעַרְכוֹת יִשְׂרָאֵל|מ2=יְיָ צְבָאוֹת אֱלֹהֵי מַעַרְכוֹת יִשְׂרָאֵל}} הֵבִֽיאוּ לִתְקוּמַת {{נוסח|הַמְּדִינָה, לִגְאֻלַּת|=מ2,פ,ר3|ר4=הָאֻמָּה וְהַמְּדִינָה וְלִגְאֻלַּת}} {{נוסח|הָעָם וְהָאָֽרֶץ|=מ2,פ,ר3|ר4=הָאָֽרֶץ}} {{נוסח|וִירוּשָׁלַֽיִם עִיר הַקֹּדֶשׁ|=מ2,פ,ר3|ר4=וְעִיר הָאֱלֹהִים}}. מִנְּשָׁרִים קַֽלּוּ וּמֵאֲרָיוֹת גָּבֵֽרוּ, בְּהֵחָלְצָם לְעֶזְרַת {{נוסח|הָעָם בַּגִּבּוֹרִים, וּבְהַנְחִילָם נִצְחוֹן עוֹלָם לְיִשְׂרָאֵל|=מ2,פ,ר3|ר4=הָעָם; וְהִרְווּ בְּדָמָם הַטָּהוֹר אֶת רִגְבֵי אַדְמַת קׇדְשֵֽׁנוּ|מ4=הָעָם; וְהִרְווּ בְּדָמָם הַטָּהוֹר אֶת רִגְבֵי אַדְמַת קׇדְשֵֽׁנוּ וּמִדְבְּרוֹת סִינָי}}. זֵֽכֶר עֲקֵדָתָם וּמַעֲשֵׂה גְבוּרָתָם לֹא יָסֽוּפוּ מֵאִתָּֽנוּ לְעוֹלָמִים. תִּהְיֶֽינָה נִשְׁמוֹתֵיהֶם צְרוּרוֹת בִּצְרוֹר הַחַיִּים עִם נִשְׁמוֹת אַבְרָהָם, יִצְחָק וְיַעֲקֹב, וְעִם {{נוסח|נִשְׁמוֹת|=מ2,פ,ר3|מ4,ר4=חסר}} שְׁאָר גִּבּוֹרֵי יִשְׂרָאֵל וּקְדוֹשָׁיו {{נוסח|שֶׁבְּגַן־עֵֽדֶן|=מ2,פ,ר3|ר4=שֶׁבְּגַן עֵֽדֶן עַד עוֹלָם}}, {{נוסח|וְנֹאמַר אָמֵן|=מ2,פ,ר3|מ4,ר4=אָמֵן}}.<קטע סוף=יזכור לחללי צה"ל מילים/>

ב. <קטע התחלה=הוראה לגבי אבינו שבשמים אל אלוהי הרוחות לכל בשר/>לזכרון חללי צה"ל (על בסיס הנוסח שהופץ ע"י משרד הדתות):{{הערה|ראו [https://www.nli.org.il/he/books/NNL_ALEPH003819462/NLI '''סדר התפלות וההודיות'''] (משרד הדתות: ירושלים תשי"ד), עמ' ג, ולאחר מכן בסידורים רבים.}}<קטע סוף=הוראה לגבי אבינו שבשמים אל אלוהי הרוחות לכל בשר/>

<קטע התחלה=אבינו שבשמים אל אלוהי הרוחות לכל בשר מילים/>אָבִֽינוּ שֶׁבַּשָּׁמַֽיִם, אֵל אֱלֹהֵי הָרוּחוֹת לְכׇל בָּשָׂר, זְכׇר־נָא אֶת הַנְּשָׁמוֹת הַזַּכּוֹת וְהַטְּהוֹרוֹת שֶׁל בָּנֵֽינוּ וּבְנוֹתֵֽינוּ אֲשֶׁר הֶעֱרוּ נַפְשָׁם לָמוּת מוֹת־גִּבּוֹרִים, בְּהֵחָלְצָם לְעֶזְרַת הָעָם וְהָאָֽרֶץ. מִנְּשָׁרִים קַֽלּוּ, וּמֵאֲרָיוֹת גָּבֵֽרוּ, בְּמִלְחַמְתָּם לְמַֽעַן שִׁחְרוּר עַמָּם וּמוֹלַדְתָּם. בַּעֲלוֹתָם עַל מִזְבַּח תְּקוּמַת יִשְׂרָאֵל בְּאֶֽרֶץ־קׇדְשׁוֹ, {{נוסח|הֵפִֽיחוּ|בסדר התפלות וההודיות "הפכו" (!)}} רֽוּחַ עֹז וּגְבוּרָה בְּכׇל בֵּית יִשְׂרָאֵל, בָּאָֽרֶץ וּבַתְּפוּצוֹת, {{נוסח|וַיִּתְעוֹרֵר|סדר התפלות וההודיות=והתעוררו (!)}} לִקְרַאת גְּאֻלָּתוֹ וּפְדוּת נַפְשׁוֹ. יִזְכְּרֵם אֱלֹהֵֽינוּ לְטוֹבָה עִם רִבְבוֹת אַלְפֵי קְדוֹשֵׁי יִשְׂרָאֵל וְגִבּוֹרָיו מִימֵי עוֹלָם. בִּצְרוֹר הַחַיִּים יִצְרֹר אֶת נִשְׁמָתָם, בְּגַן־עֵֽדֶן תְּהֵא מְנוּחָתָם, וְיָנֽוּחוּ בְּשָׁלוֹם עַל מִשְׁכָּבָם, וַיַּעַמְדוּ לְגוֹרָלָם לְקֵץ הַיָּמִין. אָמֵן.<קטע סוף=אבינו שבשמים אל אלוהי הרוחות לכל בשר מילים/>

<קטע התחלה=הוראה לאל מלא רחמים לחללי צה"ל/>אזכרה לחללי צה"ל ("אֵל מָלֵא רַחֲמִים" על בסיס הנוסח שנקבע בסידורי צה"ל בתקופתו של הרב גורן):<קטע סוף=הוראה לאל מלא רחמים לחללי צה"ל/>

<קטע התחלה=אל מלא רחמים לחללי צה"ל מילים/>אֵל מָלֵא רַחֲמִים שׁוֹכֵן בַּמְּרוֹמִים, הַמְצֵא מְנוּחָה נְכוֹנָה עַל כַּנְפֵי הַשְּׁכִינָה, בְּמַעֲלוֹת קְדוֹשִׁים, טְהוֹרִים וְגִבּוֹרִים, כְּזֹֽהַר הָרָקִֽיע {{נוסח|מַזְהִירִים|=כ,מ2,ע,פ,ק1,ק2,ר1|מְאִירִים וּמַזְהִירִים}}, {{נוסח|לְנִשְׁמוֹת הַקְּדוֹשִׁים שֶׁנִּלְחֲמוּ בְּכׇל מַעַרְכוֹת יִשְׂרָאֵל, בַּמַּחְתֶּֽרֶת וּבִצְבָא הַהֲגָנָה לְיִשְׂרָאֵל, וְשֶׁנָּפְלוּ בְּמִלְחַמְתָּם|=כ,מ2,מ3,ע,פ,ק1,ק2,ר1|מ4=חַיָּלֵי צְבָא הֲגַנָּה לְיִשְׂרָאֵל שֶׁנָּפְלוּ בְּמִלְחֶֽמֶת הַשִּׁחְרוּר, בְּמַעַרְכוֹת סִינַי וּבִפְעֻלּוֹת הֲגַנָּה}} וּמָסְרוּ נַפְשָׁם עַל קְדֻשַּׁת הַשֵּׁם, הָעָם וְהָאָֽרֶץ. בַּעֲבוּר שֶׁאָֽנוּ מִתְפַּלְּלִים לְעִלּוּי {{נוסח|נִשְׁמוֹתֵיהֶם|=כ,מ2,מ3,ע,פ,ק1,ק2,ר1|בְּעַד אַזְכָּרַת נִשְׁמוֹתֵיהֶם, בְּגַן־עֵֽדֶן מְנוּחָתָם|מ4=לְעִלּוּי נִשְׁמָתָם}}, לָכֵן בַּֽעַל הָרַחֲמִים יַסְתִּירֵם בְּסֵֽתֶר כְּנָפָיו לְעוֹלָמִים, וְיִצְרוֹר בִּצְרוֹר הַחַיִּים אֶת {{נוסח|נִשְׁמוֹתֵיהֶם|=כ,מ2,מ3,ע,פ,ק1,ק2,ר1|מ4=נִשְׁמָתָם}}. יְיָ הוּא נַחֲלָתָם, {{נוסח|בְּגַן־עֵֽדֶן תְּהֵא מְנוּחָתָם,|=מ2,ע,ק1,ק2|מ4=בְּגַן עֵֽדֶן תִּהְיֶה מְנוּחָתָם,|פ,ר3=בְּגַן עֵֽדֶן מְנוּחָתָם,}} וְיָנֽוּחוּ בְּשָׁלום עַל {{נוסח|מִשְׁכְּבוֹתָם|=מ2,ע,פ,ר1|ק1,ק2=מִשִּׁכְבוֹתֵיהֶם|מ4=מִשְׁכָּבָם}}; {{נוסח|וְתַעֲמֹד לְכׇל יִשְׂרָאֵל זְכוּתָם, וְיַעַמְדוּ לְגוֹרָלָם לְקֵץ הָיָּמִין,|=מ2,ע,פ,ק1,ק2,ר1|מ4=וְיַעַמְדוּ לְגוֹרָלָם לְקֵץ הָיָּמִין,}} וְנֹאמַר אָמֵן.<קטע סוף=אל מלא רחמים לחללי צה"ל מילים/>

<קטע התחלה=מזמור פ"ג מילים/>שִׁיר מִזְמוֹר לְאָסָף. אֱלֹהִים אַל־דֳּמִי־לָךְ, אַל־תֶּחֱרַשׁ וְאַל־תִּשְׁקֹט אֵל. כִּי הִנֵּה אוֹיְבֶֽיךָ יֶהֱמָיוּן, וּמְשַׂנְאֶֽיךָ נָֽשְׂאוּ רֹאשׁ. עַל עַמְּךָ יַעֲרִֽימוּ סוֹד, וְיִתְיָעֲצוּ עַל צְפוּנֶֽיךָ. אָמְרוּ לְכוּ וְנַכְחִידֵם מִגּוֹי, וְלֹא יִזָּכֵר שֵׁם־יִשְׂרָאֵל עוֹד. כִּי נוֹעֲצוּ לֵב יַחְדָּו, עָלֶֽיךָ בְּרִית יִכְרֹֽתוּ. אׇהֳלֵי אֱדוֹם וְיִשְׁמְעֵאלִים, מוֹאָב וְהַגְרִים. גְּבָל וְעַמּוֹן וַעֲמָלֵק, פְּלֶֽשֶׁת עִם־יֹֽשְׁבֵי צוֹר. גַּם־אַשּׁוּר נִלְוָה עִמָּם, הָיוּ זְרוֹעַ לִבְנֵי־לוֹט, סֶֽלָה. עֲשֵׂה לָהֶם כְּמִדְיָן, כְּסִיסְרָא כְיָבִין בְּנַֽחַל קִישׁוֹן. נִשְׁמְדוּ בְעֵין־דֹּאר, הָֽיוּ דֹֽמֶן לָאֲדָמָה. שִׁיתֵֽמוֹ נְדִיבֵֽימוֹ כְּעֹרֵב וְכִזְאֵב, וּכְזֶֽבַח וּכְצַלְמֻנָּע כׇּל נְסִיכֵֽימוֹ. אֲשֶׁר אָמְרוּ נִֽירְשָׁה לָּֽנוּ, אֵת נְאוֹת אֱלֹהִֽים. אֱלֹהַי שִׁיתֵֽמוֹ כַגַּלְגַּל, כְּקַשׁ לִפְנֵי־רֽוּחַ. כְּאֵשׁ תִּבְעַר־יָֽעַר, וּכְלֶהָבָה תְּלַהֵט הָרִים. כֵּן תִּרְדְּפֵם בְּסַעֲרֶֽךָ, וּבְסוּפָתְךָ תְבַהֲלֵם. מַלֵּא פְנֵיהֶם קָלוֹן, וִיבַקְשׁוּ שִׁמְךָ יְיָ. יֵבֹֽשׁוּ וְיִבָּהֲלוּ עֲדֵי־עַד, וְיַחְפְּרוּ וְיֹאבֵֽדוּ. וְיֵדְעוּ כִּי אַתָּה שִׁמְךָ יְיָ לְבַדֶּֽךָ, עֶלְיוֹן עַל כׇּל הָאָֽרֶץ.<קטע סוף=מזמור פ"ג מילים/>

<קטע התחלה=מזמור ט' מקור/>[[תהלים ט/טעמים|תהלים ט]]<קטע סוף=מזמור ט' מקור/>

<קטע התחלה=מזמור ט' מילים/>לַמְנַצֵּֽחַ עַל מוּת לַבֵּן, מִזְמוֹר לְדָוִד.  אוֹדֶה יְיָ בְּכׇל לִבִּי, אֲסַפְּרָה כׇּל נִפְלְאוֹתֶֽיךָ. אֶשְׂמְחָה וְאֶעֶלְצָה בָךְ, אֲזַמְּרָה שִׁמְךָ עֶלְיוֹן. בְּשׁוּב אוֹיְבַי אָחוֹר, יִכָּשְׁלוּ וְיֹאבְדוּ מִפָּנֶֽיךָ. כִּי עָשִֽׂיתָ מִשְׁפָּטִי וְדִינִי; יָשַֽׁבְתָּ לְכִסֵּא, שׁוֹפֵט צֶֽדֶק. גָּעַֽרְתָּ גוֹיִם, אִבַּֽדְתָּ רָשָׁע; שְׁמָם מָחִֽיתָ לְעוֹלָם וָעֶד. הָאוֹיֵב, תַּמּוּ חֳרָבוֹת לָנֶֽצַח; וְעָרִים נָתַֽשְׁתָּ, אָבַד זִכְרָם הֵֽמָּה. וַייָ לְעוֹלָם יֵשֵׁב, כּוֹנֵן לַמִּשְׁפָּט כִּסְאוֹ. וְהוּא יִשְׁפֹּט תֵּבֵל בְּצֶֽדֶק, יָדִין לְאֻמִּים בְּמֵישָׁרִים. וִיהִי יְיָ מִשְׂגָּב לַדָּךְ, מִשְׂגָּב לְעִתּוֹת בַּצָּרָה. וְיִבְטְחוּ בְךָ יוֹדְעֵי שְׁמֶֽךָ, כִּי לֹא עָזַֽבְתָּ דֹרְשֶֽׁיךָ יְיָ. זַמְּרוּ לַייָ יֹשֵׁב צִיּוֹן, הַגִּֽידוּ בָעַמִּים עֲלִילוֹתָיו. כִּי דֹרֵשׁ דָּמִים אוֹתָם זָכָר, לֹא שָׁכַח צַעֲקַת עֲנָוִים. חָנְנֵֽנִי יְיָ, רְאֵה עׇנְיִי מִשֹּׂנְאָי, מְרוֹמְמִי מִשַּֽׁעֲרֵי מָֽוֶת. לְמַֽעַן אֲסַפְּרָה כׇּל תְּהִלָּתֶֽיךָ; בְּשַׁעֲרֵי בַת צִיּוֹן אָגִֽילָה בִּישׁוּעָתֶֽךָ. טָבְעוּ גוֹיִם בְּשַֽׁחַת עָשׂוּ, בְּרֶֽשֶׁת־זוּ טָמָֽנוּ נִלְכְּדָה רַגְלָם. נוֹדַע יְיָ מִשְׁפָּט עָשָׂה; בְּפֹֽעַל כַּפָּיו נוֹקֵשׁ רָשָׁע, הִגָּיוֹן סֶֽלָה. יָשֽׁוּבוּ רְשָׁעִים לִשְׁאֽוֹלָה, כׇּל גּוֹיִם שְׁכֵחֵי אֱלֹהִים. כִּי לֹא לָנֶֽצַח יִשָּׁכַח אֶבְיוֹן, תִּקְוַת עֲנִיִּים תֹּאבַד לָעַד. קוּמָה יְיָ אַל יָעֹז אֱנוֹשׁ, יִשָּׁפְטוּ גוֹיִם עַל פָּנֶֽיךָ. שִׁיתָה יְיָ מוֹרָה לָהֶם, יֵדְעוּ גוֹיִם אֱנוֹשׁ הֵֽמָּה, סֶּֽלָה.<קטע סוף=מזמור ט' מילים/>

<קטע התחלה=אתה הראת מילים/>אַתָּה הׇרְאֵֽתָ לָדַֽעַת כִּי יְיָ הוּא הָאֱלֹהִים, אֵין עוֹד מִלְּבַדּוֹ. לְעֹשֵׂה נִפְלָאוֹת גְּדֹלוֹת לְבַדּוֹ כִּי לְעוֹלָם חַסְדּוֹ. אֵין כָּמֽוֹךָ בָאֱלֹהִים אֲדֹנָי, וְאֵין כְּמַעֲשֶֽׂיךָ. יְהִי כְבוֹד יְיָ לְעוֹלָם, יִשְׂמַח יְיָ בְּמַעֲשָׂיו. יְהִי שֵׁם יְיָ מְבֹרָךְ, מֵעַתָּה וְעַד עוֹלָם. יְהִי יְיָ אֱלֹהֵֽינוּ עִמָּֽנוּ, כַּאֲשֶׁר הָיָה עִם {{נוסח|אֲבֹתֵֽינוּ|בירנבוים=אֲבוֹתֵֽינוּ}} אַל יַעַזְבֵֽנוּ וְאַל יִטְּשֵֽׁנוּ. וְאִמְרוּ: הוֹשִׁיעֵֽנוּ אֱלֹהֵי יִשְׁעֵֽנוּ, וְקַבְּצֵֽנוּ וְהַצִּילֵֽנוּ מִן הַגּוֹיִם, {{נוסח|לְהֹדוֹת|בירנבוים=לְהוֹדוֹת}} לְשֵׁם קׇדְשֶׁךָ, לְהִשְׁתַּבֵּֽחַ בִּתְהִלָּתֶֽךָ. יְיָ מֶֽלֶךְ, יְיָ מָלָךְ, יְיָ יִמְלֹךְ לְעוֹלָם וָעֶד. יְיָ עֹז לְעַמּוֹ יִתֵּן; יְיָ יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם. וְיִהְיוּ נָא אֲמָרֵֽינוּ לְרָצוֹן לִפְנֵי אָדוֹן כֹּל. וַיְהִי בִּנְסֹֽעַ הָאָרֹן וַיֹּֽאמֶר מֹשֶׁה: קוּמָה יְיָ, וְיָפֻֽצוּ אֹיְבֶֽיךָ, {{נוסח|וְיָנֻֽסוּ|בירנבוים=וְיָנֽוּסוּ(!)}} מְשַׂנְאֶֽיךָ מִפָּנֶֽיךָ. קוּמָה יְיָ לִמְנוּחָתֶֽךָ, אַתָּה וַאֲרוֹן עֻזֶּךָ. כֹּהֲנֶֽיךָ יִלְבְּשׁוּ צֶֽדֶק, וַחֲסִידֶֽיךָ יְרַנֵּֽנוּ. בַּעֲבוּר דָּוִד עַבְדֶּֽךָ, אַל תָּשֵׁב פְּנֵי מְשִׁיחֶֽךָ. וְאָמַר בַּיּוֹם הַהוּא: הִנֵּה אֱלֹהֵֽינוּ זֶה קִוִּֽינוּ לוֹ וְיוֹשִׁיעֵֽנוּ; זֶה יְיָ קִוִּֽינוּ לוֹ, נָגִֽילָה וְנִשְׂמְחָה בִּישׁוּעָתוֹ. מַלְכוּתְךָ מַלְכוּת כׇּל עֹלָמִים, וּמֶמְשַׁלְתְּךָ בְּכׇל דּוֹר {{נוסח|וָדֹר|בירנבוים=וָדוֹר}}. כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה, וּדְבַר יְיָ מִירוּשָׁלָֽיִם.<קטע סוף=אתה הראת מילים/>
