<קטע התחלה=שלום עליכם מילים הכל/>שָׁלוֹם עֲלֵיכֶם, מַלְאֲכֵי הַשָּׁרֵת, מַלְאֲכֵי עֶלְיוֹן, מִמֶּֽלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא.{{ש}}בּוֹאֲכֶם לְשָׁלוֹם, מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מִמֶּֽלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא.{{ש}}בָּרְכֽוּנִי לְשָׁלוֹם, מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מִמֶּֽלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא.{{ש}}צֵאתְכֶם לְשָׁלוֹם, מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מִמֶּֽלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא.<קטע סוף=שלום עליכם מילים הכל/>

<קטע התחלה=כי מלאכיו/>כִּי מַלְאָכָיו {{נוסח|יְצַוֶּה־לָּךְ|בירנבוים=יְצַוֶּה לָּךְ}}, לִשְׁמׇרְךָ בְּכׇל דְּרָכֶֽיךָ. יְיָ יִשְׁמׇר־צֵאתְךָ וּבוֹאֶֽךָ, מֵעַתָּה וְעַד עוֹלָם.<קטע סוף=כי מלאכיו/>

<קטע התחלה=ריבון כל העולמים/>רִבּוֹן כׇּל הָעוֹלָמִים, אֲדוֹן כׇּל הַנְּשָׁמוֹת, אֲדוֹן הַשָּׁלוֹם, מֶֽלֶךְ אַבִּיר, מֶֽלֶךְ בָּרוּךְ, מֶֽלֶךְ גָּדוֹל, מֶֽלֶךְ דּוֹבֵר שָׁלוֹם, מֶֽלֶךְ הָדוּר, מֶֽלֶךְ וָתִיק, מֶֽלֶךְ זָךְ, מֶֽלֶךְ חֵי הָעוֹלָמִים, מֶֽלֶךְ טוֹב וּמֵטִיב, מֶֽלֶךְ יָחִיד וּמְיֻחָד, מֶֽלֶךְ כַּבִּיר, מֶֽלֶךְ לוֹבֵשׁ רַחֲמִים, מֶֽלֶךְ מַלְכֵי הַמְּלָכִים, מֶֽלֶךְ נִשְׂגָּב, מֶֽלֶךְ סוֹמֵךְ נוֹפְלִים, מֶֽלֶךְ עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית, מֶֽלֶךְ פּוֹדֶה וּמַצִּיל, מֶֽלֶךְ צַח וְאָדוֹם, מֶֽלֶךְ קָדוֹשׁ, מֶֽלֶךְ רָם וְנִשָּׁא, מֶֽלֶךְ שׁוֹמֵֽעַ תְּפִלָּה, מֶֽלֶךְ תָּמִים דַּרְכּוֹ.<קטע סוף=ריבון כל העולמים/>

<קטע התחלה=מודה אני על כל החסד מילים הכל/><קטע התחלה=מודה אני על כל החסד 1/>מוֹדֶה אֲנִי לְפָנֶֽיךָ, יְיָ אֱלֹהַי וֵאלֹהֵי אֲבוֹתַי, עַל כׇּל הַחֶֽסֶד אֲשֶׁר עָשִֽׂיתָ עִמָּדִי, וַאֲשֶׁר אַתָּה עָתִיד לַעֲשׂוֹת עִמִּי וְעִם כׇּל בְּנֵי בֵיתִי וְעִם כׇּל בְּרִיּוֹתֶֽיךָ, בְּנֵי בְרִיתִי; וּבְרוּכִים הֵם מַלְאָכֶֽיךָ הַקְּדוֹשִׁים וְהַטְּהוֹרִים שֶׁעוֹשִׂים רְצוֹנֶֽךָ. אֲדוֹן הַשָּׁלוֹם, מֶֽלֶךְ שֶׁהַשָּׁלוֹם שֶׁלּוֹ, בָּרְכֵנִי בַשָּׁלוֹם, וְתִפְקֹד אוֹתִי וְאֶת כׇּל בְּנֵי בֵיתִי, וְכׇל עַמְּךָ בֵּית יִשְׂרָאֵל, לְחַיִּים טוֹבִים וּלְשָׁלוֹם. מֶֽלֶךְ עֶלְיוֹן עַל כׇּל צְבָא מָרוֹם, יוֹצְרֵֽנוּ, יוֹצֵר בְּרֵאשִׁית, אֲחַלֶּה פָנֶֽיךָ הַמְּאִירִים, שֶׁתְּזַכֶּה אוֹתִי וְאֶת כׇּל בְּנֵי בֵיתִי לִמְצֹא חֵן וְשֵֽׂכֶל טוֹב בְּעֵינֶֽיךָ וּבְעֵינֵי כׇל בְּנֵי {{נוסח|אָדָם|=בירנבוים|אָדָם וְחַוָּה}}, וּבְעֵינֵי כׇל רוֹאֵֽינוּ, לַעֲבוֹדָתֶֽךָ. וְזַכֵּֽנִי לְקַבֵּל שַׁבָּתוֹת מִתּוֹךְ רֹב שִׂמְחָה, וּמִתּוֹךְ עֽשֶׁר וְכָבוֹד, וּמִתּוֹךְ מִעוּט עֲוֺנוֹת; וְהָסֵר מִמֶּֽנִּי וּמִכׇּל בְּנֵי בֵיתִי, וּמִכׇּל עַמְּךָ בֵּית יִשְׂרָאֵל, כׇּל מִינֵי חֹֽלִי, וְכׇל מִינֵי מַדְוֶה, וְכׇל מִינֵי דַלּוּת וַעֲנִיּוּת וְאֶבְיוֹנוּת; וְתֶן־בָּֽנוּ יֵֽצֶר טוֹב לְעׇבְדְּךָ בֶּאֱמֶת וּבְיִרְאָה וּבְאַהֲבָה. וְנִהְיֶה מְכֻבָּדִים בְּעֵינֶֽיךָ וּבְעֵינֵי כׇל רוֹאֵֽינוּ, כִּי אַתָּה הוּא מֶֽלֶךְ הַכָּבוֹד, כִּי לְךָ נָאֶה, כִּי לְךָ יָאֶה. {{נוסח|אָנָּא|אולי פסקה חדשה}}, מֶֽלֶךְ מַלְכֵי הַמְּלָכִים, צַוֵּה לְמַלְאָכֶֽיךָ, מַלְאֲכֵי הַשָּׁרֵת, מְשָׁרְתֵי עֶלְיוֹן, שֶׁיִּפְקְדֽוּנִי בְּרַחֲמִים, וִיבָרְכֽוּנִי בְּבוֹאָם לְבֵיתִי בְּיוֹם קׇדְשֵֽׁנוּ; כִּי הִדְלַֽקְתִּי נֵרוֹתַי, וְהִצַּֽעְתִּי מִטָּתִי, וְהֶחֱלַֽפְתִּי שִׂמְלוֹתַי, לִכְבוֹד יוֹם הַשַּׁבָּת, וּבָֽאתִי לְבֵיתְךָ לְהַפִּיל תְּחִנָּתִי לְפָנֶֽיךָ שֶׁתַּעֲבִיר אַנְחָתִי, וָאָעִיד אֲשֶׁר בָּרָֽאתָ בְּשִׁשָּׁה יָמִים כׇּל הַיְּצוּר; וְאֶשְׁנֶה וָאֲשַׁלֵּשׁ עוֹד לְהָעִיד עַל כּוֹסִי בְּתוֹךְ שִׂמְחָתִי, כַּאֲשֶׁר צִוִּיתַֽנִי לְזׇכְרוֹ וּלְהִתְעַנֵּג בְּיֶֽתֶר נִשְׁמָתִי אֲשֶׁר נָתַֽתָּ בִּי. בּוֹ אֶשְׁבֹּת כַּאֲשֶׁר צִוִּיתַֽנִי לְשָׁרְתֶֽךָ, וְכֵן אַגִּיד גְּדֻלָּתְךָ בְּרִנָּה. {{נוסח|וְשִׁוִּֽיתִי|=מסורה|בירנבוים=וְשִׁוִּיתִי}} יְיָ לִקְרָאתִי, שֶׁתְּרַחֲמֵֽנִי עוֹד בְּגָלוּתִי לְגׇאֳלֵֽנִי וּלְעוֹרֵר לִבִּי<קטע סוף=מודה אני על כל החסד 1/> <קטע התחלה=מודה אני על כל החסד 2/>לְאַהֲבָתֶֽךָ, וְאָז אֶשְׁמֹר {{נוסח|פִּקּוּדֶֽיךָ|=בירנבוים|לְפִקּוּדֶֽיךָ}} וְחֻקֶּֽיךָ בְּלִי עֶֽצֶב, וְאֶתְפַּלֵּל כַּדָּת, {{נוסח|כָּרָאוּי וּכְנָכוֹן|=בירנבוים|וְכָרָאוּי וְכַנָּכוֹן}}. מַלְאֲכֵי הַשָּׁלוֹם, בּוֹאֲכֶם לְשָׁלוֹם, בָּרְכֽוּנִי לְשָׁלוֹם, וְאִמְרוּ בָּרוּךְ לְשֻׁלְחָנִי הֶעָרוּךְ, וְצֵאתְכֶם לְשָׁלוֹם, מֵעַתָּה וְעַד עוֹלָם, אָמֵן סֶֽלָה.<קטע סוף=מודה אני על כל החסד 2/><קטע סוף=מודה אני על כל החסד מילים הכל/>

<קטע התחלה=אשת חיל מילים הכל/><קטע התחלה=אשת חיל 1/>אֵֽשֶׁת חַֽיִל מִי יִמְצָא וְרָחֹק מִפְּנִינִים מִכְרָהּ.{{ש}}בָּֽטַח בָּהּ לֵב בַּעְלָהּ וְשָׁלָל לֹא יֶחְסָר.{{ש}}גְּמָלַֽתְהוּ טוֹב וְלֹא רָע כֹּל יְמֵי חַיֶּֽיהָ.{{ש}}דָּרְשָׁה צֶֽמֶר וּפִשְׁתִּים וַתַּֽעַשׂ בְּחֵֽפֶץ כַּפֶּֽיהָ.{{ש}}הָיְתָה כׇּאֳנִיּוֹת סוֹחֵר מִמֶּרְחָק תָּבִיא לַחְמָהּ.{{ש}}וַתָּֽקׇם בְּעוֹד לַֽיְלָה וַתִּתֵּן טֶֽרֶף לְבֵיתָהּ{{ש}}    וְחֹק לְנַעֲרֹתֶֽיהָ.{{ש}}זָמְמָה שָׂדֶה וַתִּקָּחֵֽהוּ מִפְּרִי כַפֶּֽיהָ נָֽטְעָה כָּֽרֶם.{{ש}}חָגְרָ֣ה בְעוֹז מׇתְנֶֽיהָ וַתְּאַמֵּץ זְרוֹעֹתֶֽיהָ.{{ש}}טָעֲמָה כִּי טוֹב סַחְרָהּ לֹא יִכְבֶּה בַלַּֽיְלָה נֵרָהּ.{{ש}}יָדֶֽיהָ שִׁלְּחָה בַכִּישׁוֹר וְכַפֶּֽיהָ תָּמְכוּ פָֽלֶךְ.{{ש}}כַּפָּהּ פָּרְשָׂה לֶעָנִי וְיָדֶֽיהָ שִׁלְּחָה לָאֶבְיוֹן.{{ש}}לֹא תִירָא לְבֵיתָהּ מִשָּֽׁלֶג כִּי כׇל בֵּיתָהּ לָבֻשׁ שָׁנִים.{{ש}}מַרְבַדִּים עָשְׂתָה לָּהּ שֵׁשׁ וְאַרְגָּמָן לְבוּשָׁהּ.{{ש}}נוֹדָע בַּשְּׁעָרִים בַּעְלָהּ בְּשִׁבְתּוֹ עִם {{נוסח|זִקְנֵי־אָֽרֶץ|בירנבוים=זִקְנֵי אָֽרֶץ}}.{{ש}}סָדִין עָשְׂתָה וַתִּמְכֹּר וַחֲגוֹר נָתְנָה לַכְּנַעֲנִי.{{ש}}עֹז וְהָדָר לְבוּשָׁהּ וַתִּשְׂחַק לְיוֹם אַחֲרוֹן.<קטע סוף=אשת חיל 1/><קטע התחלה=אשת חיל 2/>{{ש}}פִּֽיהָ פָּתְחָה בְחׇכְמָה {{נוסח|וְתֽוֹרַת|=מסורה|בירנבוים=וְתוֹרַת}} חֶֽסֶד עַל לְשׁוֹנָהּ.{{ש}}צוֹפִיָּה הֲלִיכוֹת בֵּיתָהּ וְלֶֽחֶם עַצְלוּת לֹא תֹאכֵל.{{ש}}קָֽמוּ בָנֶֽיהָ וַיְאַשְּׁרֽוּהָ בַּעְלָהּ וַיְהַלְלָהּ.{{ש}}רַבּוֹת בָּנוֹת עָֽשׂוּ חָֽיִל וְאַתְּ עָלִית עַל כֻּלָּֽנָה.{{ש}}שֶֽׁקֶר הַחֵן וְהֶֽבֶל הַיֹּֽפִי אִשָּׁה יִרְאַת יְיָ הִיא תִתְהַלָּל.{{ש}}תְּנוּ לָהּ מִפְּרִי יָדֶֽיהָ וִיהַלְלֽוּהָ בַשְּׁעָרִים מַעֲשֶֽׂיהָ.<קטע סוף=אשת חיל 2/><קטע סוף=אשת חיל מילים הכל/>

==זמירות לליל שבת==

<קטע התחלה=כותרת לזמירות לליל שבת/>זְמִירוֹת לְלֵיל שַׁבָּת<קטע סוף=כותרת לזמירות לליל שבת/>

<קטע התחלה=הוראה לפני הזמירות לליל שבת/>שרים בשולחן שבת:<קטע סוף=הוראה לפני הזמירות לליל שבת/>

<קטע התחלה=כל מקדש שביעי מילים הכל/><קטע התחלה=כל מקדש שביעי 1/>כׇּל מְקַדֵּשׁ שְׁבִיעִי כָּרָאוּי לוֹ, כׇּל שׁוֹמֵר שַׁבָּת כַּדָּת מֵחַלְלוֹ, שְׂכָרוֹ הַרְבֵּה מְאֹד עַל פִּי פׇעֳלוֹ, אִישׁ עַל מַחֲנֵֽהוּ וְאִישׁ עַל דִגְלוֹ.

אוֹהֲבֵי יְיָ הַמְחַכִּים {{נוסח|לְבִנְיַן|=בירנבוים|בְּבִנְיַן}} אֲרִיאֵל, בְּיוֹם הַשַּׁבָּת שִֽׂישׂוּ וְשִׂמְחוּ כִּמְקַבְּלֵי מַתַּן נַחֲלִיאֵל, גַּם שְׂאוּ יְדֵיכֶם קֹֽדֶשׁ וְאִמְרוּ לָאֵל, בָּרוּךְ יְיָ אֲשֶׁר נָתַן מְנוּחָה לְעַמּוֹ יִשְׂרָאֵל.

דּוֹרְשֵׁי יְיָ זֶֽרַע אַבְרָהָם אוֹהֲבוֹ, הַמְאַחֲרִים לָצֵאת מִן הַשַּׁבָּת וּמְמַהֲרִים לָבֹא, שְׂמֵחִים לְשׇׁמְרוֹ וּלְעָרֵב עֵרוּבוֹ, זֶה הַיּוֹם עָשָׂה יְיָ נָגִֽילָה וְנִשְׂמְחָה בוֹ.

זִכְרוּ תּוֹרַת משֶׁה בְּמִצְוַת שַׁבָּת גְּרוּסָה, חֲרוּתָה לַיּוֹם הַשְּׁבִיעִי כְּכַלָּה בֵּין רֵעוֹתֶֽיהָ מְשֻׁבָּצָה, טְהוֹרִים יִירָשֽׁוּהָ {{נוסח|וִיקַדְּשֽׁוּהָ|בירנבוים=וִיקַדְּשֽׁוּהוּ (!)}} בְּמַאֲמַר כׇּל אֲשֶׁר עָשָׂה, וַיְכַל אֱלֹהִים בַּיּוֹם הַשְּׁבִיעִי מְלַאכְתּוֹ אֲשֶׁר עָשָׂה.

יוֹם קָדוֹשׁ הוּא מִבּוֹאוֹ וְעַד צֵאתוֹ, כׇּל זֶֽרַע יַעֲקֹב יְכַבְּדֽוּהוּ כִּדְבַר הַמֶּֽלֶךְ וְדָתוֹ, לָנֽוּחַ בּוֹ וְלִשְׂמֽוֹחַ בְּתַעֲנוּג אָכוֹל וְשָׁתוֹ, כׇּל עֲדַת יִשְׂרָאֵל יַעֲשׂוּ אוֹתוֹ.<קטע סוף=כל מקדש שביעי 1/>

<קטע התחלה=כל מקדש שביעי 2/>מְשׁוֹךְ חַסְדְּךָ לְיֹדְעֶֽיךָ אֵל קַנּוֹא וְנוֹקֵם, נוֹטְרֵי {{נוסח|יוֹם|=בירנבוים|לַיּוֹם}} הַשְּׁבִיעִי זָכוֹר וְשָׁמוֹר לְהָקֵם, שַׂמְּחֵם בְּבִנְיַן שָׁלֵם בְּאוֹר פָּנֶֽיךָ תַּבְהִיקֵם, יִרְוְיֻן מִדֶּֽשֶׁן בֵּיתֶֽךָ וְנַֽחַל עֲדָנֶֽיךָ תַשְׁקֵם.

עֲזוֹר לַשּׁוֹבְתִים בַּשְּׁבִיעִי בֶּחָרִישׁ וּבַקָּצִיר {{נוסח|לְעוֹלָמִים|=בירנבוים|עוֹלָמִים}}, פּוֹסְעִים בּוֹ פְּסִיעָה קְטַנָּה סוֹעֲדִים בּוֹ לְבָרֵךְ שָׁלֹשׁ פְּעָמִים, צִדְקָתָם תַּצְהִיר כְּאוֹר שִׁבְעַת הַיָּמִים, {{נוסח|יְיָ אֱלֹהֵי יִשְׂרָאֵל הָֽבָה תָּמִים|=בירנבוים|יְיָ אֱלֹהֵי יִשְׂרָאֵל אַהֲבַת תָּמִים, יְיָ אֱלֹהֵי יִשְׂרָאֵל תְּשׁוּעַת עוֹלָמִים}}.<קטע סוף=כל מקדש שביעי 2/><קטע סוף=כל מקדש שביעי מילים הכל/><קטע התחלה=כל מקדש שביעי הערה/>{{הערה|בסידורו הביא בירנבוים בהערה את הבית האחרון של השיר, כפי שהוא נמצא במחזור ויטרי: קַדְּשֵׁם בְּמִצְוֺתֶֽיךָ וְטַהֲרֵם כְּעֶֽצֶם הַשָּׁמַֽיִם לָטֹֽהַר, רוּחֲךָ תְּנִיחֵֽמוֹ כַּבְּהֵמָה תֵּרֵד בַּבִּקְעָה מִן הָהָר {{מקור|ישעיהו סג יד|ישעיהו סג,יד}}, שַׁבַּתָּם תְּשַׁכְּנֵם בְּנַחֲלַת {{נוסח|הַסַּֽהַר|בירנבוים=הַסַּהַר}}, כִּנְחָלִים {{נוסח|נִטָּֽיוּ|בירנבוים=נִטָּיוּ}} עֲלֵי נָהָר {{מקור|במדבר כד ו|במדבר כד,ו}}.}}<קטע סוף=כל מקדש שביעי הערה/>


<קטע התחלה=מנוחה ושמחה/>מְנוּחָה וְשִׂמְחָה אוֹר לַיְּהוּדִים, יוֹם שַׁבָּתוֹן יוֹם מַחֲמַדִּים, שׁוֹמְרָיו וְזוֹכְרָיו הֵֽמָּה מְעִידִים, כִּי לְשִׁשָּׁה כֹּל בְּרוּאִים וְעוֹמְדִים.<קטע סוף=מנוחה ושמחה/>

<קטע התחלה=במשנה לחם/>בְּמִשְׁנֶה לֶֽחֶם וְקִדּוּשׁ רַבָּה, בְּרוֹב מַטְעַמִּים וְרֽוּחַ נְדִיבָה, יִזְכּוּ {{נוסח|לְרׇב־טוּב|=בירנבוים|לְרַב טוּב}} הַמִּתְעַנְּגִים בָּהּ, בְּבִיאַת גּוֹאֵל לְחַיֵּי הָעוֹלָם הַבָּא.<קטע סוף=במשנה לחם/>

<קטע התחלה=זמירות ליום שבת כותרת משולבת/>{{נוסח|זְמִירוֹת לְ[יוֹם] שַׁבָּת|הכותרת בדומה לערב ולמוצ"ש בתוכן העניינים}}<קטע סוף=זמירות ליום שבת כותרת משולבת/>

{{#קטע:הסידור השלם (בירנבוים)/אשכנז/דפי יסוד/קידוש והבדלה וסעודות שבת ויום טוב|הוראה לפני הזמירות לליל שבת}}

<קטע התחלה=ברוך ה' יום יום מילים הכל/><קטע התחלה=ברוך ה' יום יום 1/>בָּרוּךְ אֲדֹנָי יוֹם יוֹם, יַעֲמׇס־לָֽנוּ יֶֽשַׁע וּפִדְיוֹם, וּבִשְׁמוֹ נָגִיל כׇּל הַיּוֹם, וּבִישׁוּעָתוֹ נָרִים רֹאשׁ עֶלְיוֹן, כִּי הוּא מָעוֹז לַדָּל {{נוסח|וּמַחְסֶה|=בירנבוים|וּמַחֲסֶה}} לָאֶבְיוֹן.

שִׁבְטֵי יָהּ לְיִשְׂרָאֵל עֵדוּת, בְּצָרָתָם לֹו צָר בְּסִבְלוּת וּבְעַבְדוּת, בְּלִבְנַת הַסַּפִּיר הֶרְאָם עֹז יְדִידוּת, וְנִגְלָה לְהַעֲלוֹתָם מֵעֹֽמֶק בּוֹר וָדוּת, כִּי עִם יְיָ הַחֶֽסֶד וְהַרְבֵּה עִמּוֹ פְדוּת.

מַה יָּקָר חַסְדּוֹ בְּצִלּוֹ לְגוֹנְנֵֽמוֹ, בְּגָלוּת בָּבֶֽלָה שֻׁלַּח לְמַעֲנֵֽמוֹ, לְהוֹרִיד בָּרִיחִים נִמְנָה בֵינֵֽימוֹ, וַיִּתְּנֵם לְרַחֲמִים לִפְנֵי שׁוֹבֵֽימוֹ, כִּי לֹא יִטּשׁ יְיָ אֶת עַמּוֹ, בַּעֲבוּר הַגָּדוֹל שְׁמוֹ.

עֵילָם שָׁת כִּסְאוֹ לְהַצִּיל יְדִידָיו, לְהַעֲבִיר מִשָּׁם מָעֻזְנֵי מוֹרְדָיו, מֵעֲבֹר בַּשֶּֽׁלַח פָּדָה אֶת עֲבָדָיו, קֶֽרֶן לְעַמּוֹ יָרִים תְּהִלָּה לְכׇל חֲסִידָיו, כִּי אִם הוֹגָה וְרִחַם כְּרַחֲמָיו וּכְרֹב חֲסָדָיו.

וּצְפִיר הָעִזִּים הִגְדִּיל עֲצוּמָיו, וְגַם חָזוּת אַרְבַּע עָלוּ לִמְרוֹמָיו, וּבְלִבָּם דִּמּוּ לְהַשְׁחִית אֶת רְחוּמָיו, עַל יְדֵי כֹהֲנָיו מִגֵּר מִתְקוֹמְמָיו, חַסְדֵי יְיָ כִּי לֹא תָֽמְנוּ כִּי לֹא כָלוּ רַחֲמָיו.<קטע סוף=ברוך ה' יום יום 1/>

<קטע התחלה=ברוך ה' יום יום 2/>נִסְגַּרְתִּי לֶאֱדוֹם בְּיַד רֵעַי מְדָנַי, שֶׁבְּכׇל {{נוסח|יוֹם|=בירנבוים|יוֹם וָיוֹם}} {{נוסח|מְמַלְּאִים|בירנבוים=מְמַלְאִים}} כְּרֵסָם מֵעֲדָנַי, עֶזְרָתוֹ עִמִּי לִסְמֹךְ אֶת אֲדָנַי, וְלֹא נְטַשְׁתַּֽנִי כׇּל יְמֵי עִדָּנַי, כִּי לֹא יִזְנַח לְעוֹלָם אֲדֹנָי.

בְּבוֹאוֹ מֵאֱדוֹם חֲמוּץ בְּגָדִים, זֶֽבַח לוֹ בְּבׇצְרָה וְטֶֽבַח לוֹ בְּבוֹגְדִים, וְיֵז נִצְחָם מַלְבּוּשָׁיו {{נוסח|לְהַאְדִּים|=בירנבוים|לְהַאֲדִּים}}, בְּכֹחוֹ הַגָּדוֹל יִבְצֹר רֽוּחַ נְגִידִים, הָגָה בְּרוּחוֹ הַקָּשָׁה בְּיוֹם קָדִים.

רְאוֹתוֹ כִּי כֵן אֲדוֹמִי הָעוֹצֵר, {{נוסח|יַחְשׇׁב־לוֹ|=בירנבוים|יַחֲשׁוֹב לוֹ}} {{נוסח|בׇּצְרָה|=בירנבוים|בְבָצְרָה}} תִּקְלוֹט כְּבֶֽצֶר, וּמַלְאָךְ כְּאָדָם בְּתוֹכָהּ יִנָּצֵר, וּמֵזִיד כַּשּׁוֹגֵג בְּמִקְלָט יֵעָצֵר, אֶהֱבוּ אֶת יְיָ כׇּל חֲסִידָיו אֱמוּנִים נוֹצֵר.

יְצַוֶּה צוּר חַסְדּוֹ קְהִלּוֹתָיו לְקַבֵּץ, מֵאַרְבַּע רוּחוֹת עָדָיו לְהִקָּבֵץ, וּבְהַר מְרוֹם הָרִים אוֹתָֽנוּ לְהַרְבֵּץ, וְאִתָּֽנוּ יָשׁוּב נִדָּחִים קוֹבֵץ, יָשִׁיב לֹא נֶאֱמַר כִּי אִם וְשָׁב וְקִבֵּץ.

בָּרוּךְ הוּא אֱלֹהֵֽינוּ אֲשֶׁר טוֹב גְּמָלָֽנוּ, כְּרַחֲמָיו וּכְרֹב חֲסָדָיו {{נוסח|הִגְדִּיל־לָֽנוּ|בירנבוים=הִגְדִּיל לָֽנוּ}}, אֵֽלֶּה וְכָאֵֽלֶּה יוֹסֵף עִמָּֽנוּ, לְהַגְדִּיל שְׁמוֹ הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא שֶׁנִּקְרָא עָלֵֽינוּ. בָּרוּךְ הוּא אֱלֹהֵֽינוּ שֶׁבְּרָאָֽנוּ לִכְבוֹדוֹ, {{נוסח|לְהַלְלוֹ|=בירנבוים|לְהַלְּלוֹ}} וּלְשַׁבְּחוֹ וּלְסַפֵּר הוֹדוֹ, מִכׇּל אֹם גָּבַר עָלֵֽינוּ חַסְדּוֹ, לָכֵן בְּכׇל לֵב וּבְכׇל נֶֽפֶשׁ וּבְכׇל {{נוסח|מְאוֹד|=בירנבוים|מְאוֹדוֹ}} נַמְלִיכוֹ וּנְיַחֲדוֹ. שֶׁהַשָּׁלוֹם שֶׁלּוֹ יָשִׂים עָלֵֽינוּ בְּרָכָה וְשָׁלוֹם; מִשְּׂמֹאל וּמִיָּמִין עַל יִשְׂרָאֵל שָׁלוֹם; הָרַחֲמָן הוּא יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם; וְיִזְכּוּ לִרְאוֹת בָּנִים וּבְנֵי בָנִים עוֹסְקִים בַּתּוֹרָה וּבְמִצְוֺת; עַל יִשְׂרָאֵל שָׁלוֹם; יוֹעֵץ אֵל גִּבּוֹר אֲבִי־עַד שַׂר־שָׁלוֹם.<קטע סוף=הוראה לפני זמירות ליום שבת/><קטע סוף=ברוך ה' יום יום 2/>


<קטע התחלה=ברוך אל עליון/>בָּרוּךְ אֵל עֶלְיוֹן אֲשֶׁר נָתַן מְנוּחָה, לְנַפְשֵֽׁנוּ {{נוסח|פִדְיוֹן|=בירנבוים|פִדְיוֹם}} מִשֵּׁאת וַאֲנָחָה, וְהוּא יִדְרוֹשׁ לְצִיּוֹן עִיר הַנִּדָּחָה, עַד אָֽנָה {{נוסח|תּוֹגְיוּן|=בירנבוים|תּוּגְיוֹן}} נֶֽפֶשׁ נֶאֱנָחָה.<קטע סוף=ברוך אל עליון/>

<קטע התחלה=כותרת פשוטה לזמירות למוצאי שבת/>זְמִירוֹת לְמוֹצָאֵי שַׁבָּת<קטע סוף=כותרת פשוטה לזמירות למוצאי שבת/>

<קטע התחלה=המבדיל בין קודש לחול/>הַמַּבְדִּיל בֵּין קֹֽדֶשׁ לְחֹל חַטֹּאתֵֽינוּ הוּא יִמְחֹל;{{ש}}זַרְעֵֽנוּ וְכַסְפֵּֽנוּ יַרְבֶּה כַּחוֹל וְכַכּוֹכָבִים בַּלָּֽיְלָה.

יוֹם פָּנָה כְּצֵל תֹּֽמֶר אֶקְרָא לָאֵל עָלַי גֹמֵר;{{ש}}אָמַר שׁוֹמֵר אָתָא בֹֽקֶר וְגַם לָֽיְלָה.

צִדְקָתְךָ כְּהַר תָּבוֹר עַל חֲטָאַי עָבֹר תַּעֲבֹר{{ש}}כְּיוֹם אֶתְמוֹל כִּי יַעֲבֹר וְאַשְׁמוּרָה בַּלָּֽיְלָה.

חָלְפָה עוֹנַת מִנְחָתִי מִי יִתֵּן מְנוּחָתִי;{{ש}}יָגַֽעְתִּי בְאַנְחָתִי אַשְׂחֶה בְּכׇל לָֽיְלָה.

קוֹלִי בַּל {{נוסח|יֻטַּל|=בירנבוים|יֻנְטָל}} פְּתַח לִי שַֽׁעַר הַמְּנֻטָּל;{{ש}}שֶׁרֹּאשִׁי נִמְלָא טָל קְוֻצּוֹתַי רְסִֽיסֵי לָֽיְלָה.

הֵעָתֵר נוֹרָא וְאָיוֹם אֲשַׁוֵּֽעַ תְּנָה פִדְיוֹם{{ש}}בְּנֶֽשֶׁף בְּעֶֽרֶב יוֹם בְּאִישׁוֹן לָֽיְלָה.

קְרָאתִֽיךָ יָהּ הוֹשִׁיעֵֽנִי אֹֽרַח חַיִּים תּוֹדִיעֵֽנִי{{ש}}{{נוסח|מִדַּלָּה|=בירנבוים|מִדַּלּוּת}} תְּבַצְּעֵֽנִי מִיּוֹם וְעַד לָֽיְלָה.

טַהֵר טִנּוּף מַעֲשַׂי פֶּן יֹאמְרוּ מַכְעִיסַי{{ש}}{{נוסח|אַיֵּה|=בירנבוים|אַיֵּה נָא}} אֱלֽוֹהַּ עֹשָׂי נֹתֵן זְמִרוֹת בַּלָּֽיְלָה.

נַֽחְנוּ בְיָדְךָ כַּחֹֽמֶר סְלַח נָא עַל קַל וָחֹֽמֶר;{{ש}}יוֹם לְיוֹם יַבִּֽיעַ אֹֽמֶר וְלַֽיְלָה לְלָֽיְלָה.<קטע סוף=המבדיל בין קודש לחול/>


<קטע התחלה=במוצאי יום מנוחה מילים הכל/><קטע התחלה=במוצאי יום מנוחה 1/>בְּמוֹצָאֵי יוֹם מְנוּחָה הַמְצֵא לְעַמְּךָ רְוָחָה;{{ש}}שְׁלַח תִּשְׁבִּי לְנֶאֱנָחָה וְנָס יָגוֹן וַאֲנָחָה.

{{נוסח|יָאֶה הוּא|=בירנבוים|יָאֲתָה לְךָ}} צוּרִי לְקַבֵּץ עַם מְפֻזָּרִי;{{ש}}מִיַּד גּוֹי אַכְזָרִי אֲשֶׁר כָּֽרָה לִי שׁוּחָה.

עֵת דּוֹדִים תְּעוֹרֵר אֵל לְמַלֵּט עַם אֲשֶׁר שׁוֹאֵל{{ש}}רְאוֹת טוּבְךָ בְּבוֹא גוֹאֵל לְשֶׂה פְזוּרָה נִדָּחָה.

קְרָא יֶֽשַׁע לְעַם נְדָבָה אֵל דָּגוּל מֵרְבָבָה;{{ש}}יְהִי הַשָּׁבֽוּעַ הַבָּא לִישׁוּעָה וְלִרְוָחָה.

בַּת צִיּוֹן הַשְּׁכוּלָה אֲשֶׁר הִיא הַיּוֹם גְּעוּלָה{{ש}}מְהֵרָה תִּהְיֶה בְעוּלָה אֵם הַבָּנִים שְׂמֵחָה.

מַעְיָנוֹת אֲזַי יְזוּבוּן וּפְדוּיֵי יְיָ {{נוסח|יְשׁוּבוּן|=בירנבוים|עוֹד יְשׁוּבוּן}};{{ש}}וּמֵי יֶֽשַׁע יִשְׁאֲבוּן וְהַצָּרָה נִשְׁכָּֽחָה.

נְחֵה עַמְּךָ כְּאָב רַחֲמָן יְצַפְצְפוּ עַם לֹא אַלְמָן{{ש}}דְּבַר יְיָ אֲשֶׁר נֶאֱמָן בַּהֲקִימְךָ הַבְטָחָה.

וִידִידִים {{נוסח|פְּלֵֽיטֵי|פְּלִֽיטֵי}} חֶֽרֶץ נְגִינָתָם יִפְצְחוּ בְמֶֽרֶץ;{{ש}}בְּלִי צְוָחָה וּבְלִי פֶֽרֶץ אֵין יוֹצֵאת וְאֵין צְוָחָה.

יְהִי הַחֹֽדֶשׁ הַזֶּה כִּנְבוּאַת אֲבִי חוֹזֶה;{{ש}}וְיִשָּׁמַע בְּבַֽיִת זֶה קוֹל שָׂשׂוֹן וְקוֹל שִׂמְחָה.

חָזָק יְמַלֵּא מִשְׁאֲלוֹתֵֽינוּ אַמִּיץ יַעֲשֶׂה בַּקָּשָׁתֵֽנוּ;{{ש}}וְהוּא יִשְׁלַח {{נוסח|בְּמַעֲשֵׂה|=בירנבוים|בְּכׇל מַעֲשֵׂה}} יָדֵֽינוּ בְּרָכָה וְהַצְלָחָה.<קטע סוף=במוצאי יום מנוחה 1/>

<קטע התחלה=במוצאי יום מנוחה 2/>בְּמוֹצָאֵי יוֹם גִּילָה שִׁמְךָ נוֹרָא עֲלִילָה;
{{ש}}שְׁלַח תִּשְׁבִּי לְעַם סְגֻלָּה רֶֽוַח שָׂשׂוֹן וַהֲנָחָה.

קוֹל צׇהֳלָה וְרִנָּה שְׂפָתֵֽינוּ אָז תְּרַנֵּֽנָה;
{{ש}}אָנָּא יְיָ הוֹשִֽׁיעָה נָּא אָנָּא יְיָ הַצְלִיחָה נָא.<קטע סוף=במוצאי יום מנוחה 2/><קטע סוף=במוצאי יום מנוחה מילים הכל/>

<קטע התחלה=ריבון העולמים/>רִבּוֹן הָעוֹלָמִים, אַב הָרַחֲמִים וְהַסְּלִיחוֹת, בְּסִמָּן טוֹב וּבְמַזָּל טוֹב הָחֵל עָלֵֽינוּ אֶת {{נוסח|(שֵֽׁשֶׁת)|אומרים "שֵֽׁשֶׁת" כשאין יום טוב שחל באמצע השבוע הקרוב}} יְמֵי הַמַּעֲשֶׂה הַבָּאִים לִקְרָאתֵֽנוּ לְשָׁלוֹם, חֲשׂוּכִים מִכׇּל חֵטְא וָפֶֽשַׁע, וּמְנֻקִּים מִכׇּל עָוֺן וְאַשְׁמָה וָרֶֽשַׁע, וּמְדֻבָּקִים בְּתַלְמוּד תּוֹרָה וּבְמַעֲשִׂים טוֹבִים. וְחׇנֵּֽנוּ דֵעָה בִּינָה וְהַשְׂכֵּל מֵאִתְּךָ, וְתַשְׁמִיעֵֽנוּ בָהֶם שָׂשׂוֹן וְשִׂמְחָה; וְלֹא תַעֲלֶה קִנְאָתֵֽנוּ עַל לֵב אָדָם, וְלֹא קִנְאַת אָדָם תַּעֲלֶה עַל לִבֵּנוּ. מַלְכֵּֽנוּ אֱלהֵֽינוּ, {{נוסח|הָאָב|=בירנבוים|אָב}} הָרַחֲמָן, שִׂים בְּרָכָה וּרְוָחָה וְהַצְלָחָה בְּכׇל מַעֲשֵׂה יָדֵֽינוּ; וְכׇל הַיּוֹעֵץ עָלֵינוּ וְעַל עַמְּךָ בֵּית יִשְׂרָאֵל עֵצָה טוֹבָה וּמַחֲשָׁבָה טוֹבָה, אַמְּצוֹ וּבָרְכוֹ, גַּדְּלוֹ וְקַיְּמוֹ, קַיֵּם עֲצָתוֹ, כַּדָּבָר שֶׁנֶּאֱמַר, יִתֶּן־לְךָ כִלְבָבֶֽךָ, וְכׇל עֲצָתְךָ יְמַלֵּא. וְנֶאֱמַר: {{נוסח|וְתִגְזַר־אֹֽמֶר|בירנבוים=וְתִגְזַר־אֹֽמֶר}} {{נוסח|וְיָֽקׇם־לָךְ|בירנבוים=וְיָֽקׇם לָךְ}}, וְעַל דְּרָכֶֽיךָ נָֽגַהּ אוֹר. וְכׇל הַיּוֹעֵץ עָלֵֽינוּ וְעַל כׇּל עַמְּךָ בֵּית יִשְׂרָאֵל {{נוסח|עֵצָה שֶׁאֵינָהּ טוֹבָה|עֵצָה שֶׁאֵינָהּ טוֹבָה, וּמַחֲשָׁבָה שֶׁאֵינָהּ טוֹבָה}}, תּוּפַר עֲצָתוֹ, כַּדָּבָר שֶׁנֶּאֱמַר: יְיָ הֵפִיר עֲצַת גּוֹיִם, הֵנִיא מַחְשְׁבוֹת עַמִּים. וְנֶאֱמַר: עֻֽצוּ עֵצָה וְתֻפָר, דַּבְּרוּ דָבָר וְלֹא יָקוּם, כִּי עִמָּֽנוּ אֵל.<קטע סוף=ריבון העולמים/>
