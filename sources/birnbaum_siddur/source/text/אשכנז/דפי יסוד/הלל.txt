<קטע התחלה=הריני מוכן לפני נטילת לולב/>הֲרֵֽינִי מוּכָן וּמְזֻמָּן לְקַיֵּם מִצְוַת בּוֹרְאִי, שֶׁצִּוָּֽנוּ בְּתוֹרָתוֹ: וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן פְּרִי עֵץ הָדָר, כַּפֹּת תְּמָרִים, וַעֲנַף עֵץ עָבֹת, וְעַרְבֵי־נָֽחַל. וּבְנַעֲנוּעִי אוֹתָם יַשְׁפִּֽיעַ עָלַי שֶֽׁפַע בְּרָכוֹת וּמַחְשָׁבוֹת קְדוֹשׁוֹת, שֶׁהוּא אֱלֹהֵי הָאֱלֹהִים וַאֲדֹנֵי הָאֲדֹנִים, שַׁלִּיט בְּמַֽטָּה וּבְמַֽעַל, וּמַלְכוּתוֹ בַּכֹּל מָשָֽׁלָה. וּתְהֵא חֲשׁוּבָה מִצְוַת אַרְבָּעָה מִינִים כְּאִלּוּ קִיַּמְתִּֽיהָ בְּכׇל פְּרָטֶֽיהָ וְדִקְדּוּקֶֽיהָ. יְהִי נֹֽעַם יְיָ אֱלֹהֵֽינוּ עָלֵֽינוּ; וּמַעֲשֵׂה יָדֵֽינוּ כּוֹנְנָה עָלֵֽינוּ, וּמַעֲשֵׂה יָדֵֽינוּ כּוֹנְנֵֽהוּ. בָּרוּךְ יְיָ לְעוֹלָם, אָמֵן וְאָמֵן.<קטע סוף=הריני מוכן לפני נטילת לולב/>

<קטע התחלה=בצאת ישראל מילים/>בְּצֵאת יִשְׂרָאֵל מִמִּצְרָֽיִם, בֵּית יַעֲקֹב מֵעַם לֹעֵז. הָיְתָה יְהוּדָה לְקׇדְשׁוֹ, יִשְׂרָאֵל מַמְשְׁלוֹתָיו. הַיָּם רָאָה וַיָּנֹס; הַיַּרְדֵּן יִסֹּב לְאָחוֹר. הֶהָרִים רָקְדוּ כְאֵילִים, גְּבָעוֹת כִּבְנֵי צֹאן. מַה לְּךָ הַיָּם כִּי תָנוּס; הַיַּרְדֵּן, תִּסֹּב לְאָחוֹר. הֶהָרִים, תִּרְקְדוּ כְאֵילִים; גְּבָעוֹת,  כִּבְנֵי צֹאן. מִלִּפְנֵי אָדוֹן חֽוּלִי אָֽרֶץ, מִלִּפְנֵי אֱלֽוֹהַּ יַעֲקֹב. הַהֹפְכִי הַצּוּר אֲגַם מָֽיִם, חַלָּמִישׁ {{נוסח|לְמַעְיְנוֹ־מָֽיִם|=בירנבוים וגם במסורה התיבות מוקפות}}.<קטע סוף=בצאת ישראל מילים/>

(<קטע התחלה=לא לנו מילים/><קטע התחלה=לא לנו 1/>לֹא לָנוּ, יְיָ, לֹא לָנוּ, כִּי לְשִׁמְךָ תֵּן כָּבוֹד, עַל חַסְדְּךָ, עַל אֲמִתֶּֽךָ. לָֽמָּה יֹאמְרוּ הַגּוֹיִם, אַיֵּה נָא אֱלֹהֵיהֶם. וֵאלֹהֵֽינוּ בַשָּׁמָֽיִם; כֹּל אֲשֶׁר חָפֵץ עָשָׂה. עֲצַבֵּיהֶם כֶּֽסֶף וְזָהָב, מַעֲשֵׂה יְדֵי אָדָם.<קטע סוף=לא לנו 1/> <קטע התחלה=לא לנו 2/>פֶּה לָהֶם וְלֹא יְדַבֵּֽרוּ, עֵינַֽיִם לָהֶם וְלֹא יִרְאוּ. אׇזְנַֽיִם לָהֶם וְלֹא יִשְׁמָֽעוּ, אַף לָהֶם וְלֹא יְרִיחוּן. יְדֵיהֶם וְלֹא יְמִישׁוּן, רַגְלֵיהֶם וְלֹא יְהַלֵּֽכוּ; לֹא יֶהְגּוּ בִּגְרוֹנָם. כְּמוֹהֶם יִהְיוּ עֹשֵׂיהֶם, כֹּל אֲשֶׁר בֹּטֵֽחַ בָּהֶם. יִשְׂרָאֵל, בְּטַח בַּייָ; עֶזְרָם וּמָגִנָּם הוּא. בֵּית אַהֲרֹן, בִּטְחוּ בַייָ; עֶזְרָם וּמָגִנָּם הוּא. יִרְאֵי יְיָ, בִּטְחוּ בַייָ; עֶזְרָם וּמָגִנָּם הוּא.<קטע סוף=לא לנו 2/><קטע סוף=לא לנו מילים/>)

<קטע התחלה=ה' זכרנו יברך מילים/>יְיָ זְכָרָֽנוּ יְבָרֵךְ; יְבָרֵךְ אֶת בֵּית יִשְׂרָאֵל, יְבָרֵךְ אֶת בֵּית אַהֲרֹן. יְבָרֵךְ יִרְאֵי יְיָ, הַקְּטַנִּים עִם הַגְּדֹלִים. יֹסֵף יְיָ עֲלֵיכֶם, עֲלֵיכֶם וְעַל בְּנֵיכֶם. בְּרוּכִים אַתֶּם לַייָ, עֹשֵׂה שָׁמַֽיִם וָאָֽרֶץ. הַשָּׁמַֽיִם שָׁמַֽיִם לַייָ, וְהָאָֽרֶץ נָתַן לִבְנֵי אָדָם. לֹא הַמֵּתִים יְהַלְלוּ יָהּ, וְלֹא כׇּל יֹרְדֵי דוּמָה. וַאֲנַֽחְנוּ נְבָרֵךְ יָהּ מֵעַתָּה וְעַד עוֹלָם; הַלְלוּ־יָהּ.<קטע סוף=ה' זכרנו יברך מילים/>

<קטע התחלה=למען איתן/>לְמַֽעַן אֵיתָן הַנִּזְרָק בְּלַֽהַב אֵשׁ, לְמַֽעַן בֵּן הַנֶּעֱקַד עַל עֵצִים וָאֵשׁ, לְמַֽעַן גִּבּוֹר הַנֶּאֱבַק עִם שַׂר אֵשׁ, לְמַֽעַן דְּגָלִים נָחִֽיתָ בֶּעָנָן וְאוֹר אֵשׁ, לְמַֽעַן הֹעֲלָה לַמָּרוֹם וְנִתְעַלָּה כְּמַלְאֲכֵי אֵשׁ, לְמַֽעַן וְהוּא לָךְ כְּסֶֽגֶן בְּאֶרְאֶלֵּי אֵשׁ, לְמַֽעַן זֶֽבֶד דִּבְּרוֹת הַנְּתוּנוֹת מֵאֵשׁ, לְמַֽעַן חִפּוּי יְרִיעוֹת וַעֲנַן אֵשׁ, לְמַֽעַן טֶֽכֶס הַר יָרַֽדְתָּ עָלָיו בָּאֵשׁ, לְמַֽעַן יְדִידוּת אֲשֶׁר אָהַֽבְתָּ מִשְּׁמֵי אֵשׁ, לְמַֽעַן כָּמַהּ עַד שָׁקְעָה הָאֵשׁ, לְמַֽעַן לָקַח מַחְתַּת אֵשׁ וְהֵסִיר חֲרוֹן אֵשׁ, לְמַֽעַן מְקַנֵּא קִנְאָה גְדוֹלָה בָּאֵשׁ, לְמַֽעַן נָף יָדוֹ וְיָרְדוּ אַבְנֵי אֵשׁ, לְמַֽעַן שָׂם טְלֵה חָלָב כְּלִיל אֵשׁ, לְמַֽעַן עָמַד בַּגֹּֽרֶן וְנִתְרַצָּה בָּאֵשׁ, לְמַֽעַן פִּלֵּל בָּעֲזָרָה וְיָרְדָה הָאֵשׁ, לְמַֽעַן צִיר עָלָה וְנִתְעַלָּה בְּרֶֽכֶב {{נוסח|וְסֽוּסֵי|בירנבוים=וְסוּסֵי}} אֵשׁ, לְמַֽעַן קְדוֹשִׁים מֻשְׁלָכִים בָּאֵשׁ, לְמַֽעַן רִבּוֹ רִבְבָן חָז וְנַהֲרֵי אֵשׁ, לְמַֽעַן שִׁמְמוֹת עִירְךָ הַשְּׂרוּפָה בָאֵשׁ, הוֹשַׁע נָא; לְמַֽעַן תּוֹלְדוֹת אַלּוּפֵי יְהוּדָה תָּשִׂים כְּכִיּוֹר אֵשׁ, הוֹשַׁע נָא.<קטע סוף=למען איתן/>

<קטע התחלה=לך ה' הגדולה/>לְךָ יְיָ הַגְּדֻלָּה וְהַגְּבוּרָה, וְהַתִּפְאֶֽרֶת וְהַנֵּֽצַח וְהַהוֹד, כִּי כֹל בַּשָּׁמַֽיִם וּבָאָֽרֶץ; לְךָ יְיָ הַמַּמְלָכָה וְהַמִּתְנַשֵּׂא לְכֹל לְרֹאשׁ. וְהָיָה יְיָ לְמֶלֶךְ עַל כׇּל הָאָרֶץ; בַּיּוֹם הַהוּא יִהְיֶה יְיָ אֶחָד וּשְׁמוֹ אֶחָד. וּבְתוֹרָתְךָ כָּתוּב לֵאמֹר: שְׁמַע יִשְׂרָאֵל, יְיָ אֱלֹהֵֽינוּ, יְיָ אֶחָד. בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד.<קטע סוף=לך ה' הגדולה/>

<קטע התחלה=אערוך שועי מילים הכל/><קטע התחלה=אערוך שועי 1/>'''ה''' אֶעֱרוֹךְ שׁוּעִי, בְּבֵית שַׁוְעִי, גִּלִּֽיתִי בַצּוֹם פִּשְׁעִי, דְּרַשְׁתִּֽיךָ בּוֹ לְהוֹשִׁיעִי, הַקְשִֽׁיבָה לְקוֹל שַׁוְעִי, וְקֽוּמָה וְהוֹשִׁיעִי, זְכוֹר וְרַחֵם מוֹשִׁיעִי, חַי כֵּן תְּשַׁעְשְׁעִי, טוֹב בְּאֶֽנֶק שְׁעִי, יָחִישׁ מוֹשִׁיעִי, כַּלֵּה<קטע סוף=אערוך שועי 1/> <קטע התחלה=אערוך שועי 2/>מַרְשִׁיעִי, לְבַל עוֹד תַּרְשִׁיעִי, מַהֵר אֱלֹהֵי יִשְׁעִי, נֶֽצַח לְהוֹשִׁיעִי, שָׂא נָא עֲוֺן רִשְׁעִי, עֲבוֹר עַל פִּשְׁעִי, פְּנֵה נָא לְהוֹשִׁיעִי, צוּר צַדִּיק מוֹשִׁיעִי, קַבֵּל נָא שַׁוְעִי, רוֹמֵם קֶֽרֶן יִשְׁעִי, שַׁדַּי מוֹשִׁיעִי, הוֹשַׁע נָא; תּוֹפִֽיעַ וְתוֹשִׁיעִי, הוֹשַׁע נָא.<קטע סוף=אערוך שועי 2/><קטע סוף=אערוך שועי מילים הכל/>

<קטע התחלה=אל למושעות/>'''ו''' אֵל לְמוֹשָׁעוֹת, בְּאַרְבַּע שְׁבוּעוֹת, גָּשִׁים בְּשַׁוְעוֹת, דּוֹפְקֵי עֶֽרֶך שׁוּעוֹת, הוֹגֵי שַׁעְשֻׁעוֹת, וְחִידוֹתָם מִשְׁתַּעְשְׁעוֹת, זוֹעֲקִים לְהַשְׁעוֹת, חוֹכֵי יְשׁוּעוֹת, טְפוּלִים בָּךְ שְׁעוֹת, יוֹדְעֵי בִּין שָׁעוֹת, כּוֹרְעֶֽיךָ בְּשַׁוְעוֹת, לְהָבִין שְׁמוּעוֹת, מִפִּֽיךָ נִשְׁמָעוֹת, נוֹתֵן תְּשׁוּעוֹת, סְפוּרוֹת מַשְׁמָעוֹת, עֵדוּת מַשְׁמִיעוֹת, פּוֹעֵל יְשׁוּעוֹת, צַדִּיק נוֹשָׁעוֹת, קִרְיַֽת תְּשׁוּעוֹת, רֶֽגֶשׁ תְּשׁוּאוֹת, שָׁלֹשׁ שָׁעוֹת, הוֹשַׁע נָא; תָּחִישׁ לִתְשׁוּעוֹת, הוֹשַׁע נָא.<קטע סוף=אל למושעות/>

<קטע התחלה=כהושעת אלים מילים הכל/><קטע התחלה=כהושעת אלים 1/>'''ז''' כְּהוֹשַֽׁעְתָּ אֵלִים בְּלוּד עִמָּךְ, בְּצֵאתְךָ לְיֵֽשַׁע עַמָּךְ, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ גּוֹי וֵאלֹהִים, דְּרוּשִׁים לְיֵֽשַׁע אֱלֹהִים, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ הֲמוֹן צְבָאוֹת, וְעִמָּם מַלְאֲכֵי צְבָאוֹת, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ זַכִּים מִבֵּית עֲבָדִים, חַנּוּן בְּיָדָם מַעֲבִידִים, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ טְבוּעִים בְּצוּל גְּזָרִים, יְקָרְךָ עִמָּם מַעֲבִירִים, כֵּן<קטע סוף=כהושעת אלים 1/> <קטע התחלה=כהושעת אלים 2/>הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ כַּנָּה מְשׁוֹרֶֽרֶת וַיּוֹשַׁע, לְגוֹחָהּ מְצֻיֶּֽנֶת וַיִּוָּשַׁע, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ מַאֲמַר וְהוֹצֵאתִי אֶתְכֶם, נָקוּב וְהוּצֵאתִי אִתְּכֶם, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ סוֹבְבֵי מִזְבֵּֽחַ, עוֹמְסֵי עֲרָבָה לְהַקִּיף מִזְבֵּֽחַ, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ פִּלְאֵי אָרוֹן כְּהֻפְשַׁע, צִעֵר פְּלֶֽשֶׁת בַּחֲרוֹן אַף וְנוֹשַׁע, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ קְהִלּוֹת בָּבֶֽלָה שִׁלַּחְתָּ, רַחוּם לְמַעֲנָם שֻׁלַּחְתָּ, כֵּן הוֹשַׁע נָא. כְּהוֹשַֽׁעְתָּ שְׁבוּת שִׁבְטֵי יַעֲקֹב, תָּשׁוּב וְתָשִׁיב שְׁבוּת אׇהֳלֵי יַעֲקֹב, וְהוֹשִֽׁיעָה נָּא. כְּהוֹשַֽׁעְתָּ שׁוֹמְרֵי מִצְוֺת, וְחוֹכֵי יְשׁוּעוֹת, אֵל לְמוֹשָׁעוֹת, וְהוֹשִֽׁיעַ נָא.<קטע סוף=כהושעת אלים 2/><קטע סוף=כהושעת אלים מילים הכל/>

אָנָּא גְאַל כַּנַּת נִטְעָךְ,{{ש}}דּוּמָה {{נוסח|בְּטַאְטְאָךְ|בירנבוים=בְּטַאטְאָךְ}}, וְהוֹשִֽׁיעָה נָּא.

אָנָּא הַבֵּט לִבְרִית טִבְעָךְ,{{ש}}וּמַחֲשַׁכֵּי אֶֽרֶץ בְּהַטְבִּיעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא זְכׇר־לָֽנוּ אָב יְדָעָךְ,{{ש}}חַסְדְּךָ לָֽמוֹ בְּהוֹדִיעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא טְהֽוֹרֵי לֵב בְּהַפְלִיאָךְ,{{ש}}יִוָּדַע כִּי הוּא פִלְאָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא כַּבִּיר כֹּֽחַ תֵּן לָֽנוּ יִשְׁעָךְ,{{ש}}לַאֲבוֹתֵֽינוּ כְּהִשָּׁבְעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא מַלֵּא מִשְׁאֲלוֹת עַם מְשַׁוְּעָךְ,{{ש}}נֶעֱקָד כְּמוֹ בְּהַר מוֹר שִׁוְּעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא שַׂגֵּב אֶשְׁלֵי נִטְעָךְ,{{ש}}עָרִיצִים בַּהֲנִיעָךְ, וְהוֹשִֽׁיעָה נָּא.<קטע סוף=אנא אזון חין 1/>

<קטע התחלה=אנא אזון חין 2/>אָנָּא פְּתַח לָֽנוּ אוֹצְרוֹת רִבְעָךְ,{{ש}}צִיָּה מֵהֶם בְּהַרְבִּיעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא קוֹרְאֶֽיךָ אֶֽרֶץ בְּרוֹעֲעָךְ,{{ש}}רְעֵם בְּטוּב מִרְעָךְ, וְהוֹשִֽׁיעָה נָּא.

אָנָּא שְׁעָרֶֽיךָ תַּֽעַל מִמְּשׁוֹאָךְ,{{ש}}תֵּל תַּלְפִּיּוֹת בְּהַשִּׂיאָךְ, וְהוֹשִֽׁיעָה נָּא.<קטע סוף=אנא אזון חין 2/><קטע סוף=אנא אזון חין מילים הכל/>

<קטע התחלה=אנא אל נא הושע נא והושיעה נא מילים/>אָנָּא אֵל נָא, הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.<קטע סוף=אנא אל נא הושע נא והושיעה נא מילים/>

<קטע התחלה=אל נא תעינו מילים הכל/><קטע התחלה=אל נא תעינו 1/>אֵל נָא תָּעִֽינוּ כְּשֶׂה אוֹבֵד, שְׁמֵֽנוּ מִסִּפְרְךָ אַל תְּאַבֵּד,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא רְעֵה אֶת צֹאן הַהֲרֵגָה, קְצוּפָה וְעָלֶֽיךָ הֲרוּגָה,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא צֹאנְךָ וְצֹאן מַרְעִיתֶֽךָ, פְּעֻלָּתְךָ וְרַעְיָתֶֽךָ,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא עֲנִיֵּי הַצֹּאן, שִׂיחָם עֲנֵה בְּעֵת רָצוֹן,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא נוֹשְׂאֵי לְךָ עַֽיִן, מִתְקוֹמְמֶֽיךָ יִהְיוּ כְאַֽיִן,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא לִמְנַסְּכֵי לְךָ מַֽיִם, כְּמִמַּעַיְנֵי הַיְשׁוּעָה יִשְׁאֲבוּן מַֽיִם,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא יַעֲלוּ לְצִיּוֹן מוֹשִׁיעִים, טְפוּלִים בְּךָ וּבְשִׁמְךָ נוֹשָׁעִים,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא חֲמוּץ בְּגָדִים, זְעוֹם לְנַעֵר כׇּל בּוֹגְדִים,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.<קטע סוף=אל נא תעינו 1/>

<קטע התחלה=אל נא תעינו 2/>אֵל נָא וְזָכוֹר תִּזְכּוֹר, הַנְּכוּרִים בְּלֶֽתֶךְ וָכוֹר,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא דּוֹרְשֶֽׁיךָ בְּעַנְפֵי עֲרָבוֹת, גַּעְיָם שְׁעֵה מֵעֲרָבוֹת,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.

אֵל נָא בָּרֵךְ בְּעִטּוּר שָׁנָה, אֲמָרַי רְצֵה בְּפִלּוּלִי בְּיוֹם הוֹשַׁעְנָא,{{ש}} הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא.<קטע סוף=אל נא תעינו 2/><קטע סוף=אל נא תעינו מילים הכל/>

<קטע התחלה=הוראה לפני אנא אל נא הושע נא והושיעה נא אבינו אתה/>ש"ץ וקהל:<קטע סוף=הוראה לפני אנא אל נא הושע נא והושיעה נא אבינו אתה/>

<קטע התחלה=אנא אל נא הושע נא והושיעה נא אבינו אתה מילים/>אָנָּא אֵל נָא, הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.<קטע סוף=אנא אל נא הושע נא והושיעה נא אבינו אתה מילים/>

<קטע התחלה=למען תמים בדורותיו מילים הכל/><קטע התחלה=למען תמים בדורותיו 1/>לְמַֽעַן תָּמִים בְּדוֹרוֹתָיו, הַנִּמְלָט בְּרֹב צִדְקוֹתָיו, מֻצָּל מִשֶּֽׁטֶף בְּבוֹא מַבּוּל מַֽיִם, לְאֹם אֲנִי חוֹמָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן שָׁלֵם בְּכׇל מַעֲשִׂים, הַמְנֻסֶּה בַּעֲשָׂרָה נִסִּים, כְּשָׁר מַלְאָכִים נָם יֻקַּח נָא מְעַט מַֽיִם, לְבָרָה כַּחַמָּה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן רַךְ וְיָחִיד נֶחְנַט פְּרִי לְמֵאָה, זָעַק וְאַיֵּה הַשֶּׂה לְעוֹלָה, בִּשְּׂרֽוּהוּ עֲבָדָיו מָצָֽאנוּ מַֽיִם לְגוֹלָה וְסוּרָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן קִדֵּם שְׂאֵת בְּרָכָה, הַנִּשְׂטָם וּלְשִׁמְךָ חִכָּה, מְיַחֵם בְּמַקְלוֹת בְּשִׁקֲתוֹת הַמַּֽיִם. לְדָמְתָה לְתָמָר הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן צָדַק הֱיוֹת לְךָ לְכֹהֵן, כֶּחָתָן פְּאֵר יְכַהֵן, מְנֻסֶּה בְּמַסָּה<קטע סוף=למען תמים בדורותיו 1/> <קטע התחלה=למען תמים בדורותיו 2/>בְּמֵי מְרִֽיבַת מַֽיִם, לְהָהָר הַטּוֹב הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן פֹּאַר הֱיוֹת גְּבִיר לְאֶחָיו, יְהוּדָה אֲשֶׁר גָּבַר בְּאֶחָיו, מִסְפַּר רֹֽבַע מִדָּלְיָו יִזַּל מַֽיִם, לֹא לָֽנוּ כִּי אִם לְמַעַנְךָ הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן עָנָו מִכֹּל וְנֶאֱמָן, אֲשֶׁר בְּצִדְקוֹ כִּלְכֵּל הַמָּן, מָשׁוּךְ לְגוֹאֵל וּמָשׁוּי מִמַּֽיִם, לְמִי זֹאת הַנִּשְׁקָפָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן שַׂמְתּוֹ כְּמַלְאֲכֵי מְרוֹמִים, הַלּוֹבֵשׁ אוּרִים וְתֻמִּים, מְצֻוֶּה לָבֹא בַּמִּקְדָּשׁ בְּקִדּוּשׁ יָדַֽיִם וְרַגְלַֽיִם וּרְחִֽיצַת מַֽיִם, לְחוֹלַת אַהֲבָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן נְבִיאָה מְחוֹלַת מַחֲנַֽיִם, לִכְמֵֽהֵי לֵב הוּשְׂמָה עֵינַֽיִם, לְרַגְלָהּ רָֽצָה עֲלוֹת וָרֶֽדֶת בְּאֵר מַֽיִם, לְטֽוֹבוּ אֹהָלָיו הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן מְשָׁרֵת לֹא מָשׁ מֵאֹֽהֶל, וְרֽוּחַ הַקֹּֽדֶשׁ עָלָיו אִהֵל, בְּעׇבְרוֹ בַיַּרְדֵּן נִכְרְתוּ הַמַּֽיִם, לְיָפָה וּבָרָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן לִמֵּד רְאוֹת לְטוֹבָה אוֹת, זָעַק אַיֵּה נִפְלָאוֹת, מִצָּה טַל מִגִּזָּה מְלֹא הַסֵּֽפֶל מַֽיִם, לְכַלַּת לְבָנוֹן הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן כְּלוּלֵי עֲשׂוֹת מִלְחַמְתֶּֽךָ, אֲשֶׁר בְּיָדָם תַּֽתָּה יְשׁוּעָתֶֽךָ, צְרוּפֵי מִגּוֹי בְּלָקְקָם בְּיָדָם מַֽיִם, לְלֹא בָגְדוּ בָךְ הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.<קטע סוף=למען תמים בדורותיו 2/>

<קטע התחלה=למען תמים בדורותיו 3/>לְמַֽעַן יָחִיד צוֹרְרִים דָּשׁ, אֲשֶׁר מֵרֶֽחֶם לְנָזִיר הֻקְדָּשׁ, מִמַּכְתֵּשׁ לֶֽחִי הִבְקַֽעְתָּ לּוֹ מַֽיִם, לְמַֽעַן שֵׁם קׇדְשֶֽׁךָ הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן טוֹב הוֹלֵךְ וְגָדֵל, אֲשֶׁר מֵעֹֽשֶׁק עֵדָה חִדֵּל, בְּשׁוּב עַם מֵחֵטְא צָו שְׁאׇב־מַֽיִם, לְנָאוָה כִּירוּשָׁלַֽיִם הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן חַיָּךְ מְכַרְכֵּר בְּשִׁיר. הַמְלַמֵּד תּוֹרָה בְּכׇל כְּלֵי שִׁיר. מְנַסֵּךְ לְפָנֶֽיךָ כְּתָאַב שְׁתוֹת מַֽיִם, לְשָֽׂמוּ בְךָ סִבְרָם הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן זָךְ עָלָה בַסְּעָרָה, הַמְקַנֵּא וּמֵשִׁיב עֶבְרָה, לְפִלּוּלוֹ יָרְדָה אֵשׁ וְלִחֲכָה עָפָר וּמַֽיִם, לְעֵינֶֽיהָ בְּרֵכוֹת הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן וְשֵׁרֵת בֶּאֱמֶת לְרַבּוֹ, פִּי שְׁנַֽיִם בְּרוּחוֹ נֶאֱצַל בּוֹ, בְּקַחְתּוֹ מְנַגֵּן נִתְמַלְּאוּ גֵבִים מַֽיִם, לְפָצוּ מִי כָמֹֽכָה הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן הִרְהֵר עֲשׂוֹת רְצוֹנֶֽךָ, הַמַּכְרִיז תְּשׁוּבָה לְצֹאנֶֽךָ, אָז בְּבוֹא מְחָרֵף סָתַם עֵינוֹת מַֽיִם, לְצִיּוֹן מִכְלַל יֹֽפִי הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן דְּרָשֽׁוּךָ בְּתוֹךְ הַגּוֹלָה, וְסוֹדְךָ לָֽמוֹ נִגְלָה, בְּלִי לְהִתְגָּאֵל דָּרְשׁוּ זֵרְעוֹנִים וּמַֽיִם, לְקוֹרְאֶֽיךָ בַצָּר הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

לְמַֽעַן גָּמַר חׇכְמָה וּבִינָה, סוֹפֵר מָהִיר מְפַלֵּשׂ אֲמָנָה, מְחַכְּמֵֽנוּ אֲמָרִים הַמְּשׁוּלִים בְּרַחֲבֵי מַֽיִם, לְרַבָּֽתִי עָם הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.<קטע סוף=למען תמים בדורותיו 3/>

<קטע התחלה=למען תמים בדורותיו 4/>לְמַֽעַן בָּאֵי לְךָ הַיּוֹם בְּכׇל לֵב, שׁוֹפְכִים לְךָ שִֽׂיחַ בְּלֹא לֵב וָלֵב, שׁוֹאֲלִים מִמְּךָ עֹז מִטְרוֹת מַֽיִם, לְשׁוֹרְרוּ בַיָּם הוֹשַׁע נָא וְהוֹשִֽׁיעָה נָּא, אָבִֽינוּ אָֽתָּה.

<קטע התחלה=אומץ ישעך בא מילים הכל/><קטע התחלה=אומץ ישעך בא 1/>אֹֽמֶץ יֶשְׁעֲךָ בָּא, קוֹל דּוֹדִי הִנֵּה זֶה בָּא, מְבַשֵּׂר וְאוֹמֵר.

קוֹל בָּא בְּרִבְבוֹת כִּתִּים, לַעֲמוֹד עַל הַר הַזֵּיתִים, מְבַשֵּׂר וְאוֹמֵר.

קוֹל גִּשְׁתּוֹ בַּשּׁוֹפָר לִתְקַע, תַּחְתָּיו הַר יִבָּקַע, מְבַשֵּׂר וְאוֹמֵר.

קוֹל דָּפַק וְהֵצִיץ וְזָרָח, וּמָשׁ חֲצִי הָהָר מִמִּזְרָח, מְבַשֵּׂר וְאוֹמֵר.

קוֹל הֵקִים מִלּוּל נׇאֳמוֹ, וּבָא הוּא וְכׇל קְדוֹשִׁים עִמּוֹ, מְבַשֵּׂר וְאוֹמֵר.<קטע סוף=אומץ ישעך בא 1/>

<קטע התחלה=אומץ ישעך בא 2/>קוֹל וּלְכׇל בָּאֵי עוֹלָם, בַּת קוֹל יִשָּׁמַע בָּעוֹלָם, מְבַשֵּׂר וְאוֹמֵר.

קוֹל זֶֽרַע עֲמוּסֵי רַחְמוֹ, נוֹלְדוּ כְּיֶֽלֶד מִמְּעֵי אִמּוֹ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל חָֽלָה וְיָלְדָה מִי זֹאת, מִי שָׁמַע כָּזֹאת, מְבַשֵּׂר וְאוֹמֵר.

קוֹל טָהוֹר פָּעַל כׇּל אֵֽלֶּה, וּמִי רָאָה כָּאֵֽלֶּה, מְבַשֵּׂר וְאוֹמֵר.

קוֹל יֶֽשַׁע וּזְמַן הוּחַד, הֲיֽוּחַל אֶֽרֶץ בְּיוֹם אֶחָד, מְבַשֵּׂר וְאוֹמֵר.

קוֹל כַּבִּיר רוֹם וָתַֽחַת, אִם יִוָּלֵד גּוֹי פַּֽעַם אֶחָת, מְבַשֵּׂר וְאוֹמֵר.

קוֹל לְעֵת יִגְאַל עַמּוֹ נָאוֹר, וְהָיָה לְעֵת עֶֽרֶב יִהְיֶה אוֹר, מְבַשֵּׂר וְאוֹמֵר.

קוֹל מוֹשִׁיעִים יַעֲלוּ לְהַר צִיּוֹן, כִּי חָֽלָה גַּם יָלְדָה צִיּוֹן, מְבַשֵּׂר וְאוֹמֵר.

קוֹל נִשְׁמַע בְּכׇל גְּבוּלֵךְ, הַרְחִֽיבִי מְקוֹם אׇהֳלֵךְ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל שִֽׂימִי עַד דַּמֶּֽשֶׂק מִשְׁכְּנוֹתַֽיִךְ, קַבְּלִי בָנַֽיִךְ וּבְנוֹתַֽיִךְ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל עִלְזִי חֲבַצֶּֽלֶת הַשָּׁרוֹן, כִּי קָֽמוּ יְשֵׁנֵי חֶבְרוֹן, מְבַשֵּׂר וְאוֹמֵר.

קוֹל פְּנוּ אֵלַי וְהִוָּשְׁעוּ, הַיּוֹם אִם בְּקוֹלִי תִשְׁמָֽעוּ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל צָמַח אִישׁ צֶֽמַח שְׁמוֹ, הוּא דָּוִד בְּעַצְמוֹ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל קֽוּמוּ כְּפוּשֵׁי עָפָר, הָקִֽיצוּ וְרַנְּנוּ שׁוֹכְנֵי עָפָר, מְבַשֵּׂר וְאוֹמֵר.

קוֹל רַבָּֽתִי עָם בְּהַמְלִיכוֹ, מִגְדּוֹל יְשׁוּעוֹת מַלְכּוֹ, מְבַשֵּׂר וְאוֹמֵר.

קוֹל שֵׁם רְשָׁעִים לְהַאֲבִיד, עֽוֹשֶׂה חֶֽסֶד לִמְשִׁיחוֹ לְדָוִד, מְבַשֵּׂר וְאוֹמֵר.

קוֹל תְּנָה יְשׁוּעוֹת לְעַם עוֹלָם, לְדָוִד וּלְזַרְעוֹ עַד עוֹלָם, מְבַשֵּׂר וְאוֹמֵר.<קטע סוף=אומץ ישעך בא 2/><קטע סוף=אומץ ישעך בא מילים הכל/>

<קטע התחלה=יהי רצון אחרי חבטת הערבות/>יְהִי רָצוֹן מִלְפָנֶֽיךָ, יְיָ אֱלֹהֵֽינוּ וֵאלֹהֵי אֲבוֹתֵֽינוּ, הַבּוֹחֵר בִּנְבִיאִים טוֹבִים וּבְמִנְהֲגֵיהֶם הַטּוֹבִים, שֶׁתְּקַבֵּל בְּרַחֲמִים וּבְרָצוֹן אֶת תְּפִלָּתֵֽנוּ וַהֲקָפוֹתֵֽינוּ, וְתִזְכׇּר־לָֽנוּ זְכוּת שִׁבְעַת תְּמִימֶֽיךָ, וְתָסִיר מְחִיצַת הַבַּרְזֶל הַמַּפְסֶֽקֶת בֵּינֵֽינוּ וּבֵינֶֽיךָ, וְתַאֲזִין שַׁוְעָתֵֽנוּ וְתֵיטִיב לָֽנוּ הַחֲתִימָה, {{נוסח|תּֽוֹלֶה|בירנבוים=תּוֹלֶה}} אֶֽרֶץ עַל בְּלִימָה; {{נוסח|וְחׇתְמֵֽנוּ בְּסֵֽפֶר חַיִים טוֹבִים.|אחרי זה השמיט ברינבוים: "וְהַיּוֹם הַזֶה תִּתֵּן בִּשְׁכִינַת עֻזֶּֽךָ חֲמִשָּׁה גְּבוּרוֹת מְמֻתָּקוֹת עַל יְדֵי חֲבִיטַת עֲרָבָה, מִנְהַג נְבִיאֶֽיךָ הַקְּדוֹשִׁים; וְתִתְעוֹרֵר הָאַהֲבָה בֵּינֵיהֶם, וּתְנַשְּׁקֵֽנוּ בִּנְּשִׁיקַת פִּֽיךָ מְמַתֶּֽקֶת כׇּל הַגְּבוּרוֹת וְכׇל הַדִּינִין; וְתָאִיר לִשְׁכִינַת עֻזֶּֽךָ בְּשֵׁם (יו"ד ה"א וא"ו) שֶׁהוּא טַל אוֹרוֹת טַלֶּֽךָ. וּמִשָּׁם תַּשְׁפִּֽיעַ שֶֽׁפַע לְעַבְדְךָ הַמִתְפַּלְל לְפָנֶֽיךָ, מְחִילָה שֶׁתַּאֲרִיךְ יָמַי, וְתִמְחוֹל לִי חֲטָאַי וַעֲוֺנוֹתַי וּפְשָׁעַי, וְתִפְשׁוֹט יְמִינְךָ וְיָדֶֽךָ לְקַבְּלֵֽנִי בִּתְשׁוּבָה שְׁלֵמָה לְפָנֶֽיךָ." (השלמנו על בסיס סידור ברכות והודעות, עמ' 339; <https://archive.org/details/sidurberakhoteho00newyuoft/page/n4>.)}} וְאוֹצָרְךָ הַטּוֹב תִּפְתַּח לְהַשְׂבִּֽיעַ מַֽיִם נֶֽפֶשׁ שׁוֹקֵקָה, כְּמוֹ שֶׁכָּתוּב: יִפְתַּח יְיָ לְךָ אֶת אוֹצָרוֹ הַטּוֹב, אֶת הַשָּׁמַיִם, לָתֵת מְטַר אַרְצְךָ בְּעִתּוֹ, וּלְבָרֵךְ אֶת כׇּל מַעֲשֵׂה יָדֶךָ. אָמֵן.<קטע סוף=יהי רצון אחרי חבטת הערבות/>
